חֵלֶק רִאשׁוֹן: קִטְעֵי קְרִיאָה וּשְׁאֵלוֹת

קִרְאוּ אֶת קֶטַע הַמֵּידָע שֶׁלִּפְנֵיכֶם וַעֲנוּ עַל הַשְּׁאֵלוֹת שֶׁאַחֲרָיו.

הַלֵּב
  1. קִפְצוּ אֶת כַּף הַיָּד לְאֶגְרוֹף. זֶה בְּעֵרֶךְ הַגּוֹדֶל שֶׁל הַלֵּב שֶׁלָּכֶם.
  2. הַלֵּב נִמְצָא בְּתוֹךְ הֶחָזֶה, כִּמְעַט בָּאֶמְצַע, וְנוֹטֶה מְעַט לְצַד שְׂמֹאל.
  3. הַלֵּב הוּא שְׁרִיר, כְּמוֹ הַשְּׁרִירִים שֶׁבַּיָּדַיִים וּבָרַגְלַיִים.
  4. אֲבָל שְׁרִירֵי הָרַגְלַיִים מִתְעַיְּיפִים וְנָחִים, וְהַלֵּב אֵינוֹ נָח אַף פַּעַם.
  5. הוּא עוֹבֵד בַּיּוֹם וּבַלַּיְלָה, גַּם כְּשֶׁאֲנַחְנוּ יְשֵׁנִים שֵׁינָה עֲמוּקָה.
  6. מָה הָעֲבוֹדָה שֶׁל הַלֵּב? הוּא מִתְכַּוֵּוץ וְדוֹחֵף אֶת הַדָּם אֶל כָּל הַגּוּף.
  7. הַדָּם זוֹרֵם בְּתוֹךְ צִינּוֹרוֹת דַּקִּים, וּמֵבִיא לְכָל אֵיבָר חַמְצָן וּמָזוֹן.
  8. כָּל הִתְכַּוְּוצוּת שֶׁל הַלֵּב נִקְרֵאת פְּעִימָה.
  9. כְּשֶׁאַתֶּם יוֹשְׁבִים בְּשֶׁקֶט, הַלֵּב שֶׁלָּכֶם פּוֹעֵם בְּעֵרֶךְ 90 פְּעָמִים בְּדַקָּה.
  10. כְּשֶׁאַתֶּם רָצִים אוֹ קוֹפְצִים, הַשְּׁרִירִים זְקוּקִים לְהַרְבֵּה יוֹתֵר חַמְצָן.
  11. לָכֵן הַלֵּב מַתְחִיל לִפְעוֹם מַהֵר יוֹתֵר, וְשׁוֹלֵחַ אֲלֵיהֶם עוֹד וְעוֹד דָּם.
  12. אַחֲרֵי כַּמָּה דַּקּוֹת שֶׁל מְנוּחָה, הַלֵּב חוֹזֵר לִפְעוֹם בַּקֶּצֶב הָרָגִיל.
  13. רוֹצִים לְהַרְגִּישׁ אֶת הַלֵּב שֶׁלָּכֶם עוֹבֵד? אֵין צוֹרֶךְ בְּמַכְשִׁיר מְיוּחָד.
  14. הַנִּיחוּ שְׁתֵּי אֶצְבָּעוֹת עַל שׁוֹרֶשׁ כַּף הַיָּד, מִתַּחַת לָאֲגוּדָל.
  15. לַחֲצוּ בַּעֲדִינוּת, וְתַרְגִּישׁוּ נְקִישׁוֹת קְטַנּוֹת. לַנְּקִישׁוֹת הָאֵלֶּה קוֹרְאִים דּוֹפֶק.
  16. כָּל נְקִישָׁה כָּזֹאת הִיא פְּעִימָה אַחַת שֶׁל הַלֵּב שֶׁלָּכֶם.

עִבְרוּ לָעַמּוּד הַבָּא.

כָּעֵת, עֲנוּ עַל הַשְּׁאֵלוֹת שֶׁלִּפְנֵיכֶם.
הַקִּיפוּ אֶת הַתְּשׁוּבָה הַנְּכוֹנָה בְּכָל שְׁאֵלָה.

1. מַדּוּעַ הַלֵּב פּוֹעֵם מַהֵר יוֹתֵר בִּזְמַן רִיצָה?
  1. א.כִּי בִּזְמַן רִיצָה הַלֵּב מִתְעַיֵּיף וְרוֹצֶה לָנוּחַ
  2. ב.כִּי בִּזְמַן רִיצָה הַדָּם מַפְסִיק לִזְרוֹם בַּצִּינּוֹרוֹת
  3. ג.כִּי בִּזְמַן רִיצָה הַשְּׁרִירִים זְקוּקִים לְיוֹתֵר חַמְצָן
  4. ד.כִּי בִּזְמַן רִיצָה הַלֵּב גָּדֵל וְנַעֲשֶׂה גָּדוֹל מֵאֶגְרוֹף
2. בְּשׁוּרָה 11 כָּתוּב: "וְשׁוֹלֵחַ אֲלֵיהֶם עוֹד וְעוֹד דָּם".
לְמִי אוֹ לְמָה מִתְיַיחֶסֶת הַמִּילָּה "אֲלֵיהֶם"?
  1. א.לַשְּׁרִירִים
  2. ב.לַיְּלָדִים הַיְּשֵׁנִים
  3. ג.לָאֶצְבָּעוֹת וְלָאֲגוּדָל
  4. ד.לַחַמְצָן וְלַמָּזוֹן
3. מָה אֶפְשָׁר לִלְמוֹד מִשּׁוּרוֹת 13–16?
  1. א.שֶׁהַלֵּב נִמְצָא בְּשׁוֹרֶשׁ כַּף הַיָּד, מִתַּחַת לָאֲגוּדָל
  2. ב.שֶׁרַק רוֹפֵא עִם מַכְשִׁיר מְיוּחָד יָכוֹל לִבְדּוֹק אֶת הַדּוֹפֶק
  3. ג.שֶׁהַלֵּב פּוֹעֵם מַהֵר יוֹתֵר כְּשֶׁלּוֹחֲצִים עַל כַּף הַיָּד
  4. ד.שֶׁכָּל אֶחָד יָכוֹל לְהַרְגִּישׁ אֶת פְּעִימוֹת הַלֵּב שֶׁלּוֹ בְּעֶזְרַת הָאֶצְבָּעוֹת
4. עַל אֵיזוֹ שְׁאֵלָה אֵין תְּשׁוּבָה בַּקֶּטַע?
  1. א.מָה מֵבִיא הַדָּם אֶל הָאֵיבָרִים שֶׁבַּגּוּף?
  2. ב.כַּמָּה דָּם יֵשׁ בַּגּוּף שֶׁל יֶלֶד?
  3. ג.אֵיפֹה בַּגּוּף נִמְצָא הַלֵּב?
  4. ד.אֵיךְ אֶפְשָׁר לְהַרְגִּישׁ אֶת הַדּוֹפֶק?
5. הַשְׁלִימוּ אֶת הַמִּשְׁפָּט לְפִי הַקֶּטַע. כִּתְבוּ מִילָּה אַחַת עַל הַקַּו.
הַלֵּב שֶׁל כָּל אֶחָד מֵאִיתָּנוּ הוּא בְּעֵרֶךְ בַּגּוֹדֶל שֶׁל .

עִבְרוּ לָעַמּוּד הַבָּא.

קִרְאוּ אֶת הַסִּיפּוּר שֶׁלִּפְנֵיכֶם וַעֲנוּ עַל הַשְּׁאֵלוֹת שֶׁאַחֲרָיו.

הַפֶּתֶק הַיָּשָׁן
  1. בְּיוֹם שִׁישִּׁי בַּבּוֹקֶר בָּאָה אֵלָה אֶל הַבַּיִת שֶׁל סָבְתָא שׁוֹשַׁנָּה.
  2. סָבְתָא הוֹצִיאָה מֵהַמְּגֵירָה פֶּתֶק יָשָׁן וְצָהוֹב, מְקוּפָּל לְאַרְבָּעָה.
  3. "זֶה הַמַּתְכּוֹן לָעוּגָה שֶׁל אִימָּא שֶׁלִּי, סָבְתָא רַבְּתָא שֶׁלָּךְ," אָמְרָה.
  4. "הִיא כָּתְבָה אוֹתוֹ לִפְנֵי שָׁנִים רַבּוֹת, וַאֲנִי שׁוֹמֶרֶת עָלָיו מֵאָז."
  5. אֵלָה קָרְאָה: "שָׁלוֹשׁ בֵּיצִים, כּוֹס סוּכָּר, שְׁתֵּי כּוֹסוֹת קֶמַח, וַחֲצִי כּוֹס מִיץ..."
  6. כָּאן נִמְרַח הַדְּיוֹ, וְאִי אֶפְשָׁר הָיָה לִקְרוֹא אֵיזֶה מִיץ כָּתוּב שָׁם.
  7. "אוּלַי נָכִין עוּגָה אַחֶרֶת?" הִצִּיעָה אֵלָה. אֲבָל סָבְתָא לֹא הִסְכִּימָה.
  8. "לֹא מְוַותְּרִים כָּל כָּךְ מַהֵר! נְנַסֶּה לְגַלּוֹת בְּעַצְמֵנוּ מָה הָיָה כָּתוּב שָׁם."
  9. בַּנִּיסָּיוֹן הָרִאשׁוֹן הֵן סָחֲטוּ לִימוֹן. הָעוּגָה יָצְאָה חֲמוּצָה מְאוֹד.
  10. סָבְתָא טָעֲמָה וְעִיוְּותָה אֶת פָּנֶיהָ. "לֹא, זֶה לֹא הַטַּעַם שֶׁאֲנִי זוֹכֶרֶת."
  11. בַּנִּיסָּיוֹן הַשֵּׁנִי הֵן שָׁפְכוּ מִיץ תַּפּוּחִים. הָעוּגָה יָצְאָה תְּפֵלָה, כִּמְעַט בְּלִי טַעַם.
  12. אֵלָה הִבִּיטָה בַּעֲרֵמַת הַכֵּלִים שֶׁבַּכִּיּוֹר וְנֶאֶנְחָה. "אוּלַי בֶּאֱמֶת נַפְסִיק?"
  13. הִיא נִיגְּשָׁה אֶל הַחַלּוֹן, וְהִבְחִינָה בְּעֵץ גָּדוֹל בֶּחָצֵר, עָמוּס תַּפּוּזִים.
  14. "סָבְתָא, הָעֵץ הַזֶּה הָיָה כָּאן כְּשֶׁהָיִית יַלְדָּה?" שָׁאֲלָה פִּתְאוֹם.
  15. "בְּוַודַּאי. אִימָּא שֶׁלִּי שָׁתְלָה אוֹתוֹ, וְקָטְפָה מִמֶּנּוּ תַּפּוּזִים כָּל חוֹרֶף."
  16. עֵינֶיהָ שֶׁל אֵלָה נָצְצוּ. "אָז אוּלַי הִיא שָׂמָה בָּעוּגָה מִיץ תַּפּוּזִים?"
  17. הֵן קָטְפוּ שְׁלוֹשָׁה תַּפּוּזִים, סָחֲטוּ אוֹתָם, וְהִכְנִיסוּ עוּגָה שְׁלִישִׁית לַתַּנּוּר.
  18. כַּעֲבוֹר חֲצִי שָׁעָה הִתְמַלֵּא הַמִּטְבָּח בְּרֵיחַ מָתוֹק.
  19. סָבְתָא טָעֲמָה פְּרוּסָה, עָצְמָה אֶת עֵינֶיהָ, וְדִמְעָה קְטַנָּה זָלְגָה עַל לֶחְיָהּ.
  20. "זֶה בְּדִיּוּק הַטַּעַם," לָחֲשָׁה. "כְּאִילּוּ אִימָּא שֶׁלִּי עוֹמֶדֶת כָּאן בַּמִּטְבָּח."
  21. אֵלָה לָקְחָה עֵט, וְכָתְבָה עַל פֶּתֶק חָדָשׁ אֶת הַמִּילָּה הַחֲסֵרָה.
  22. "גַּם אֲנִי אֶשְׁמוֹר עַל הַמַּתְכּוֹן," אָמְרָה, "וְיוֹם אֶחָד אָכִין אוֹתוֹ עִם הַנְּכָדִים שֶׁלִּי."

עִבְרוּ לָעַמּוּד הַבָּא.

כָּעֵת, עֲנוּ עַל הַשְּׁאֵלוֹת שֶׁלִּפְנֵיכֶם.
הַקִּיפוּ אֶת הַתְּשׁוּבָה הַנְּכוֹנָה בְּכָל שְׁאֵלָה.

1. מָה פֵּירוּשׁ הַמִּילָּה "תְּפֵלָה" (שׁוּרָה 11)?
  1. א.מְתוּקָה מְאוֹד
  2. ב.חַסְרַת טַעַם
  3. ג.שְׂרוּפָה
  4. ד.קְטַנָּה מִדַּי
2. בְּשׁוּרָה 12 כָּתוּב שֶׁאֵלָה הִבִּיטָה בַּעֲרֵמַת הַכֵּלִים שֶׁבַּכִּיּוֹר.
הַמַּטָּרָה שֶׁל הַמִּשְׁפָּט הַזֶּה הִיא –
  1. א.לְהַסְבִּיר שֶׁסָּבְתָא אֵינָהּ אוֹהֶבֶת לִשְׁטוֹף כֵּלִים
  2. ב.לְהַרְאוֹת שֶׁהַמִּטְבָּח שֶׁל סָבְתָא קָטָן מְאוֹד
  3. ג.לְסַפֵּר שֶׁאֵלָה בָּאָה כְּדֵי לַעֲזוֹר בִּשְׁטִיפַת הַכֵּלִים
  4. ד.לְהַדְגִּישׁ שֶׁהֵן כְּבָר עָבְדוּ הַרְבֵּה, וְלָכֵן אֵלָה כִּמְעַט הִתְיָיאֲשָׁה
3. מָה קָרָה לִפְנֵי שֶׁאֵלָה הִבְחִינָה בָּעֵץ שֶׁבֶּחָצֵר?
  1. א.הָעוּגָה יָצְאָה חֲמוּצָה מְאוֹד
  2. ב.הֵן קָטְפוּ שְׁלוֹשָׁה תַּפּוּזִים
  3. ג.דִּמְעָה זָלְגָה עַל לֶחְיָהּ שֶׁל סָבְתָא
  4. ד.אֵלָה כָּתְבָה עַל פֶּתֶק חָדָשׁ
