חֵלֶק רִאשׁוֹן: קִטְעֵי קְרִיאָה וּשְׁאֵלוֹת

קִרְאוּ אֶת הַהוֹרָאוֹת שֶׁלִּפְנֵיכֶם וַעֲנוּ עַל הַשְּׁאֵלוֹת שֶׁאַחֲרֵיהֶן.

אֵיךְ מְגַדְּלִים שְׁעוּעִית בְּצִנְצֶנֶת?
  1. רוֹצִים לִרְאוֹת בְּמוֹ עֵינֵיכֶם אֵיךְ זֶרַע קָטָן הוֹפֵךְ לְצֶמַח יָרוֹק?
  2. לֹא צָרִיךְ גִּינָּה וְגַם לֹא עָצִיץ. מַסְפִּיקָה צִנְצֶנֶת אַחַת לְיַד הַחַלּוֹן.
  3. מָה צָרִיךְ לְהָכִין? צִנְצֶנֶת זְכוּכִית שְׁקוּפָה, צֶמֶר גֶּפֶן, מְעַט מַיִם
  4. וּשְׁלוֹשָׁה זְרָעִים שֶׁל שְׁעוּעִית יְבֵשָׁה. וְעַכְשָׁיו – לַעֲבוֹדָה!
  5. שָׁלָב 1: מַשְׁרִים אֶת הַזְּרָעִים בְּכוֹס מַיִם לְמֶשֶׁךְ לַיְלָה שָׁלֵם.
  6. שָׁלָב 2: מְרַפְּדִים אֶת תַּחְתִּית הַצִּנְצֶנֶת בְּשִׁכְבָה עָבָה שֶׁל צֶמֶר גֶּפֶן.
  7. שָׁלָב 3: מַרְטִיבִים אֶת צֶמֶר הַגֶּפֶן בִּמְעַט מַיִם, עַד שֶׁהוּא לַח.
  8. שָׁלָב 4: מַנִּיחִים אֶת הַזְּרָעִים בֵּין צֶמֶר הַגֶּפֶן וּבֵין דּוֹפֶן הַצִּנְצֶנֶת,
  9. כָּךְ שֶׁאֶפְשָׁר יִהְיֶה לִרְאוֹת אוֹתָם מִבַּחוּץ.
  10. שָׁלָב 5: מַעֲמִידִים אֶת הַצִּנְצֶנֶת לְיַד הַחַלּוֹן, וּמְחַכִּים בְּסַבְלָנוּת.
  11. כַּמָּה עֵצוֹת חֲשׁוּבוֹת: בִּדְקוּ אֶת צֶמֶר הַגֶּפֶן בְּכָל יוֹם. אִם הוּא יָבֵשׁ,
  12. הוֹסִיפוּ לוֹ כַּמָּה טִיפּוֹת מַיִם. הִיזָּהֲרוּ לֹא לְהָצִיף אֶת הַצִּנְצֶנֶת בְּמַיִם!
  13. זְרָעִים שֶׁשּׁוֹכְבִים בְּתוֹךְ שְׁלוּלִית אֵינָם נוֹבְטִים, אֶלָּא נִרְקָבִים.
  14. הַזְּרָעִים צְרִיכִים אוֹר, אֲבָל לֹא שֶׁמֶשׁ יְשִׁירָה. שֶׁמֶשׁ חֲזָקָה מִדַּי
  15. מְיַיבֶּשֶׁת אֶת צֶמֶר הַגֶּפֶן בִּמְהִירוּת וַעֲלוּלָה לִפְגּוֹעַ בַּנְּבָטִים הָעֲדִינִים.
  16. אַחֲרֵי שְׁלוֹשָׁה אוֹ אַרְבָּעָה יָמִים יִבְקַע מִכָּל זֶרַע שׁוֹרֶשׁ לָבָן וְקָטָן.
  17. כַּמָּה יָמִים אַחַר כָּךְ יִצְמַח גַּם גִּבְעוֹל יָרוֹק, וְעָלָיו שְׁנֵי עָלִים רִאשׁוֹנִים.
  18. כְּשֶׁהַצֶּמַח יַגִּיעַ לְגוֹבַהּ שְׂפַת הַצִּנְצֶנֶת, הַעֲבִירוּ אוֹתוֹ לְעָצִיץ עִם אֲדָמָה.

עִבְרוּ לָעַמּוּד הַבָּא.

כָּעֵת, עֲנוּ עַל הַשְּׁאֵלוֹת שֶׁלִּפְנֵיכֶם.
בִּשְׁאֵלוֹת 1–4 הַקִּיפוּ אֶת הַתְּשׁוּבָה הַנְּכוֹנָה. בִּשְׁאֵלָה 5 כִּתְבוּ אֶת הַתְּשׁוּבָה עַל הַקַּו.

1. מָה עוֹשִׂים מִיָּד אַחֲרֵי שֶׁמְּרַפְּדִים אֶת תַּחְתִּית הַצִּנְצֶנֶת בְּצֶמֶר גֶּפֶן?
  1. א.מַשְׁרִים אֶת הַזְּרָעִים בְּכוֹס מַיִם
  2. ב.מַנִּיחִים אֶת הַזְּרָעִים בַּצִּנְצֶנֶת
  3. ג.מַרְטִיבִים אֶת צֶמֶר הַגֶּפֶן בִּמְעַט מַיִם
  4. ד.מַעֲמִידִים אֶת הַצִּנְצֶנֶת לְיַד הַחַלּוֹן
2. מָה פֵּירוּשׁ הַמִּילָּה "לַח" (שׁוּרָה 7)?
  1. א.רָטוֹב מְעַט
  2. ב.יָבֵשׁ מְאוֹד
  3. ג.לָבָן וְנָקִי
  4. ד.עָבֶה וְרַךְ
3. יֶלֶד רוֹצֶה לְגַדֵּל שְׁעוּעִית בְּצִנְצֶנֶת לְפִי הַהוֹרָאוֹת שֶׁבַּקֶּטַע.
אֵיפֹה כְּדַאי לוֹ לְהַעֲמִיד אֶת הַצִּנְצֶנֶת?
  1. א.בְּתוֹךְ אָרוֹן סָגוּר וְחָשׁוּךְ
  2. ב.לְיַד חַלּוֹן מוּאָר, בְּמָקוֹם שֶׁקַּרְנֵי הַשֶּׁמֶשׁ אֵינָן מַגִּיעוֹת אֵלָיו
  3. ג.עַל אֶדֶן הַחַלּוֹן, בְּמָקוֹם שֶׁהַשֶּׁמֶשׁ זוֹרַחַת עָלָיו כָּל הַיּוֹם
  4. ד.בַּמְּקָרֵר, כְּדֵי שֶׁצֶּמֶר הַגֶּפֶן לֹא יִתְיַיבֵּשׁ
4. בְּשׁוּרָה 13 כָּתוּב שֶׁזְּרָעִים שֶׁשּׁוֹכְבִים בְּתוֹךְ שְׁלוּלִית אֵינָם נוֹבְטִים, אֶלָּא נִרְקָבִים.
הַמַּטָּרָה שֶׁל הַמִּשְׁפָּט הַזֶּה הִיא –
  1. א.לְסַפֵּר שֶׁשְּׁעוּעִית גְּדֵלָה בִּשְׁלוּלִיּוֹת
  2. ב.לְהַדְגִּישׁ שֶׁהַזְּרָעִים צְרִיכִים הַרְבֵּה אוֹר
  3. ג.לְשַׁכְנֵעַ אוֹתָנוּ לְהַשְׁקוֹת אֶת הַזְּרָעִים פְּעָמִים רַבּוֹת
  4. ד.לְהַסְבִּיר מַדּוּעַ אָסוּר לְהָצִיף אֶת הַצִּנְצֶנֶת בְּמַיִם
5. הַשְׁלִימוּ אֶת הַמִּשְׁפָּט לְפִי הַקֶּטַע. כִּתְבוּ מִילָּה אַחַת עַל הַקַּו.

הַדָּבָר הָרִאשׁוֹן שֶׁיּוֹצֵא מִן הַזֶּרַע הוּא .

עִבְרוּ לָעַמּוּד הַבָּא.

קִרְאוּ אֶת הַסִּיפּוּר שֶׁלִּפְנֵיכֶם וַעֲנוּ עַל הַשְּׁאֵלוֹת שֶׁאַחֲרָיו.

אַלּוֹן רוֹצֶה כֶּלֶב
  1. אַלּוֹן בֶּן הַשְּׁמוֹנֶה חָלַם עַל דָּבָר אֶחָד בִּלְבַד: כֶּלֶב מִשֶּׁלּוֹ.
  2. "כֶּלֶב הוּא לֹא צַעֲצוּעַ," אָמְרָה אִימָּא. "צָרִיךְ לְטַפֵּל בּוֹ בְּכָל יוֹם."
  3. "אֲנִי אֲטַפֵּל בּוֹ! אֲנִי מַבְטִיחַ!" קָרָא אַלּוֹן.
  4. אַבָּא חָשַׁב רֶגַע וְאָמַר: "הַבְטָחוֹת קַל לָתֵת. בּוֹא נִרְאֶה מַעֲשִׂים.
  5. הַשְּׁכֵנָה צִילָה נוֹסַעַת לְשָׁבוּעַ. אִם תְּטַפֵּל בַּכֶּלֶב שֶׁלָּהּ, שׁוֹקוֹ,
  6. שָׁבוּעַ שָׁלֵם בְּלִי שֶׁנַּזְכִּיר לְךָ – נְדַבֵּר עַל כֶּלֶב."
  7. אַלּוֹן הִסְכִּים מִיָּד. "שָׁבוּעַ? זֶה קַלֵּי קַלּוּת!" אָמַר בְּבִיטָּחוֹן.
  8. בִּשְׁלוֹשֶׁת הַיָּמִים הָרִאשׁוֹנִים הוּא קָם מוּקְדָּם, הֶאֱכִיל אֶת שׁוֹקוֹ וְיָצָא אִיתּוֹ לְטִיּוּל.
  9. בַּיּוֹם הָרְבִיעִי יָרַד גֶּשֶׁם שׁוֹטֵף. אַלּוֹן הֵצִיץ מֵהַחַלּוֹן וְנֶאֱנַח.
  10. "אוּלַי הַיּוֹם שׁוֹקוֹ יְוַותֵּר עַל הַטִּיּוּל?" שָׁאַל בְּשֶׁקֶט.
  11. אִימָּא וְאַבָּא לֹא עָנוּ. הֵם רַק הִבִּיטוּ זֶה בָּזֶה.
  12. אַלּוֹן הִבִּיט בְּשׁוֹקוֹ. הַכֶּלֶב עָמַד לְיַד הַדֶּלֶת וְכִשְׁכֵּשׁ בִּזְנָבוֹ.
  13. "טוֹב, בְּסֵדֶר," מִלְמֵל אַלּוֹן, לָבַשׁ מְעִיל וְיָצָא אִיתּוֹ אֶל הַגֶּשֶׁם.
  14. בַּיּוֹם הַחֲמִישִׁי הִזְמִינוּ אוֹתוֹ הַחֲבֵרִים לְמִשְׂחַק כַּדּוּרֶגֶל בַּמִּגְרָשׁ.
  15. אַלּוֹן כְּבָר רָץ אֶל הַדֶּלֶת, וְאָז נֶעֱצַר. "קוֹדֶם שׁוֹקוֹ," אָמַר לְעַצְמוֹ.
  16. הוּא טִיֵּיל עִם הַכֶּלֶב, וְרַק אַחַר כָּךְ הִצְטָרֵף לַמִּשְׂחָק.
  17. בַּיּוֹם הַשְּׁבִיעִי חָזְרָה צִילָה, וְשׁוֹקוֹ קָפַץ עָלֶיהָ. אַלּוֹן הִרְגִּישׁ צְבִיטָה בַּלֵּב.
  18. "אָז מָה אַתָּה אוֹמֵר עַכְשָׁיו?" שָׁאַל אַבָּא. "עֲדַיִין קַלֵּי קַלּוּת?"
  19. "דַּוְוקָא לֹא," הוֹדָה אַלּוֹן. "זֶה הָיָה קָשֶׁה. אֲפִילּוּ בַּגֶּשֶׁם צָרִיךְ לָצֵאת.
  20. אֲבָל כְּשֶׁשּׁוֹקוֹ שָׂמַח לִקְרָאתִי, שָׁכַחְתִּי אֶת כָּל הַקּוֹשִׁי."
  21. אִימָּא חִיְּיכָה. "זֹאת הַתְּשׁוּבָה שֶׁחִיכִּינוּ לָהּ," אָמְרָה.
  22. "בְּשַׁבָּת נִיסַּע יַחַד לִבְחוֹר כֶּלֶב – וְאַתָּה תִּבְחַר לוֹ שֵׁם."

עִבְרוּ לָעַמּוּד הַבָּא.

כָּעֵת, עֲנוּ עַל הַשְּׁאֵלוֹת שֶׁלִּפְנֵיכֶם.
הַקִּיפוּ אֶת הַתְּשׁוּבָה הַנְּכוֹנָה בְּכָל שְׁאֵלָה.

1. מַדּוּעַ בִּיקֵּשׁ אַבָּא מֵאַלּוֹן לְטַפֵּל בַּכֶּלֶב שֶׁל הַשְּׁכֵנָה?
  1. א.כִּי הַשְּׁכֵנָה צִילָה בִּיקְּשָׁה מֵאַלּוֹן עֶזְרָה
  2. ב.כְּדֵי לִרְאוֹת אִם אַלּוֹן בֶּאֱמֶת מְסוּגָּל לְטַפֵּל בְּכֶלֶב בְּכָל יוֹם
  3. ג.כִּי אַלּוֹן פָּחַד מִכְּלָבִים, וְאַבָּא רָצָה שֶׁיִּתְרַגֵּל אֲלֵיהֶם
  4. ד.כְּדֵי שֶׁאַלּוֹן יִשְׁכַּח שֶׁהוּא רוֹצֶה כֶּלֶב מִשֶּׁלּוֹ
2. בְּשׁוּרָה 9 כָּתוּב: "אַלּוֹן הֵצִיץ מֵהַחַלּוֹן".
אֵיזוֹ מִילָּה יְכוֹלָה לְהַחְלִיף אֶת הַמִּילָּה "הֵצִיץ"?
  1. א.קָפַץ
  2. ב.צָעַק
  3. ג.יָצָא
  4. ד.הִסְתַּכֵּל
3. בְּשׁוּרָה 19 אַלּוֹן אוֹמֵר: "אֲפִילּוּ בַּגֶּשֶׁם צָרִיךְ לָצֵאת".
מַדּוּעַ הוּא אוֹמֵר "אֲפִילּוּ"?
