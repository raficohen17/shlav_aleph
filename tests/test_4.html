חֵלֶק רִאשׁוֹן: קִטְעֵי קְרִיאָה וּשְׁאֵלוֹת

קִרְאוּ אֶת קֶטַע הַמֵּידָע שֶׁלִּפְנֵיכֶם וַעֲנוּ עַל הַשְּׁאֵלוֹת שֶׁאַחֲרָיו.

פִּינְגְּוִין הַקֵּיסָר
  1. בְּאַנְטַארְקְטִיקָה, הַיַּבֶּשֶׁת הַקָּרָה בְּיוֹתֵר בָּעוֹלָם, חַי עוֹף מְיוּחָד: פִּינְגְּוִין הַקֵּיסָר.
  2. זֶהוּ הַפִּינְגְּוִין הַגָּדוֹל בְּיוֹתֵר, וְגוֹבְהוֹ כִּמְעַט כְּגוֹבַהּ שֶׁל יֶלֶד בְּכִיתָּה ב'.
  3. רוֹב הַצִּיפּוֹרִים מְטִילוֹת בֵּיצִים בָּאָבִיב, כְּשֶׁמֶּזֶג הָאֲוִויר נָעִים וְחָמִים.
  4. אֲבָל אִימָּא פִּינְגְּוִין מְטִילָה בֵּיצָה דַּוְוקָא בְּאֶמְצַע הַחוֹרֶף, כְּשֶׁהַקּוֹר נוֹרָא.
  5. הִיא מְטִילָה בֵּיצָה אַחַת בִּלְבַד, וּמִיָּד מַעֲבִירָה אוֹתָהּ בִּזְהִירוּת אֶל הָאַבָּא.
  6. אַחַר כָּךְ הָאִימָּא יוֹצֵאת לְמַסָּע אָרוֹךְ אֶל הַיָּם, כְּדֵי לֶאֱכוֹל דָּגִים וּלְהִתְחַזֵּק.
  7. הָאַבָּא נִשְׁאָר לִשְׁמוֹר עַל הַבֵּיצָה. הוּא מַנִּיחַ אוֹתָהּ עַל כַּפּוֹת רַגְלָיו
  8. וּמְכַסֶּה אוֹתָהּ בְּקֶפֶל עוֹר חַם. אִם הַבֵּיצָה תִּיגַּע בַּקֶּרַח, הִיא תִּקְפָּא בִּמְהֵרָה.
  9. בְּמֶשֶׁךְ חוֹדְשַׁיִים הָאַבָּא עוֹמֵד כָּךְ בַּקּוֹר וּבָרוּחַ, וְאֵינוֹ אוֹכֵל דָּבָר!
  10. כְּדֵי לְהִתְחַמֵּם, כָּל הָאֲבוֹת נִצְמָדִים זֶה לָזֶה בִּקְבוּצָה גְּדוֹלָה וּצְפוּפָה.
  11. הֵם גַּם מִתְחַלְּפִים בַּמְּקוֹמוֹת: מִי שֶׁעָמַד בַּחוּץ, בָּרוּחַ, נִכְנָס לְהִתְחַמֵּם בָּאֶמְצַע.
  12. כַּעֲבוֹר חוֹדְשַׁיִים הַבֵּיצָה בּוֹקַעַת, וְגוֹזָל קָטָן וְאָפוֹר מֵצִיץ מִתּוֹכָהּ.
  13. בְּדִיּוּק אָז הָאִימָּא חוֹזֶרֶת מִן הַיָּם, וּבְבִטְנָהּ מָזוֹן רַב בִּשְׁבִיל הַגּוֹזָל.
  14. אֵיךְ הִיא מוֹצֵאת אֶת מִשְׁפַּחְתָּהּ בֵּין אַלְפֵי פִּינְגְּוִינִים שֶׁנִּרְאִים אוֹתוֹ דָּבָר?
  15. הִיא מְזַהָה אֶת הַקּוֹל שֶׁל הָאַבָּא! לְכָל פִּינְגְּוִין יֵשׁ קְרִיאָה מְיוּחֶדֶת מִשֶּׁלּוֹ.
  16. עַכְשָׁיו הָאִימָּא שׁוֹמֶרֶת עַל הַגּוֹזָל, וְהָאַבָּא הָרָעֵב יוֹצֵא סוֹף־סוֹף אֶל הַיָּם.

עִבְרוּ לָעַמּוּד הַבָּא.

כָּעֵת, עֲנוּ עַל הַשְּׁאֵלוֹת שֶׁלִּפְנֵיכֶם.
הַקִּיפוּ אֶת הַתְּשׁוּבָה הַנְּכוֹנָה בְּכָל שְׁאֵלָה.

1. מַדּוּעַ הָאַבָּא מַנִּיחַ אֶת הַבֵּיצָה עַל כַּפּוֹת רַגְלָיו?
  1. א.כְּדֵי שֶׁהָאִימָּא תְּזַהֶה אוֹתוֹ כְּשֶׁתַּחֲזוֹר מִן הַיָּם
  2. ב.כְּדֵי שֶׁיּוּכַל לָלֶכֶת אִיתָּהּ אֶל הַיָּם וְלֶאֱכוֹל דָּגִים
  3. ג.כְּדֵי שֶׁהַבֵּיצָה לֹא תִּיגַּע בַּקֶּרַח וְתִקְפָּא
  4. ד.כְּדֵי שֶׁהַבֵּיצָה תִּהְיֶה גְּדוֹלָה יוֹתֵר
2. בְּשׁוּרָה 4 כָּתוּב שֶׁאִימָּא פִּינְגְּוִין מְטִילָה בֵּיצָה דַּוְוקָא בְּאֶמְצַע הַחוֹרֶף.
מַדּוּעַ כָּתוּב "דַּוְוקָא"?
  1. א.כִּי זֶה מַפְתִּיעַ: רוֹב הַצִּיפּוֹרִים מְטִילוֹת בֵּיצִים כְּשֶׁחָמִים
  2. ב.כִּי הַפִּינְגְּוִינִים אוֹהֲבִים אֶת הַחוֹרֶף יוֹתֵר מֵאֶת הַקַּיִץ
  3. ג.כִּי בְּאֶמְצַע הַחוֹרֶף יֵשׁ בַּיָּם הַרְבֵּה דָּגִים
  4. ד.כִּי בַּחוֹרֶף הָאַבָּא אֵינוֹ אוֹכֵל דָּבָר
3. בְּשׁוּרָה 7 כָּתוּב: "הוּא מַנִּיחַ אוֹתָהּ עַל כַּפּוֹת רַגְלָיו".
לְמִי אוֹ לְמָה מִתְיַיחֶסֶת הַמִּילָה "אוֹתָהּ"?
  1. א.לָאִימָּא
  2. ב.לַבֵּיצָה
  3. ג.לָרֶגֶל
  4. ד.לַיַּבֶּשֶׁת
4. מָה אֶפְשָׁר לִלְמוֹד מִשּׁוּרוֹת 10–11?
  1. א.שֶׁהָאֲבוֹת רָבִים זֶה עִם זֶה עַל הַמָּקוֹם שֶׁבָּאֶמְצַע
  2. ב.שֶׁבְּאֶמְצַע הַקְּבוּצָה קַר יוֹתֵר מֵאֲשֶׁר בַּחוּץ
  3. ג.שֶׁהָאִימָּהוֹת מְחַמְּמוֹת אֶת הָאֲבוֹת בַּחוֹרֶף
  4. ד.שֶׁהָאֲבוֹת עוֹזְרִים זֶה לָזֶה לְהִתְגַּבֵּר עַל הַקּוֹר
5. עַל אֵיזוֹ שְׁאֵלָה אֵין תְּשׁוּבָה בַּקֶּטַע?
  1. א.כַּמָּה בֵּיצִים מְטִילָה אִימָּא פִּינְגְּוִין?
  2. ב.כַּמָּה זְמַן שׁוֹמֵר הָאַבָּא עַל הַבֵּיצָה?
  3. ג.כַּמָּה שׁוֹקֵל פִּינְגְּוִין הַקֵּיסָר?
  4. ד.אֵיךְ הָאִימָּא מוֹצֵאת אֶת הָאַבָּא כְּשֶׁהִיא חוֹזֶרֶת?

עִבְרוּ לָעַמּוּד הַבָּא.

קִרְאוּ אֶת הַסִּיפּוּר שֶׁלִּפְנֵיכֶם וַעֲנוּ עַל הַשְּׁאֵלוֹת שֶׁאַחֲרָיו.

יוֹם הַהוּלֶּדֶת שֶׁל שִׁירָן
  1. הַבּוֹקֶר קָמָה שִׁירָן נִרְגֶּשֶׁת מְאוֹד: הַיּוֹם יוֹם הַהוּלֶּדֶת הַשְּׁמִינִי שֶׁלָּהּ!
  2. בַּכִּיתָּה שֶׁל הַמּוֹרָה דַּפְנָה, כָּל יֶלֶד שֶׁחוֹגֵג יוֹם הוּלֶּדֶת מְקַבֵּל כֶּתֶר,
  3. וְכָל הַכִּיתָּה שָׁרָה לוֹ שִׁיר כְּבָר בַּשִּׁיעוּר הָרִאשׁוֹן.
  4. לְיַד הַשַּׁעַר פָּגְשָׁה שִׁירָן אֶת גִּלְעָד, שֶׁחָזַר לְבֵית הַסֵּפֶר אַחֲרֵי שָׁבוּעַ שֶׁל מַחֲלָה.
  5. כְּשֶׁנִּכְנְסוּ לַכִּיתָּה, רָאֲתָה שִׁירָן שְׁלוֹשָׁה יְלָדִים מִתְלַחֲשִׁים לְיַד הַלּוּחַ.
  6. כְּשֶׁהִתְקָרְבָה הֵם הִשְׁתַּתְּקוּ, וְאַחַת הַבָּנוֹת הִסְתִּירָה דַּף צִבְעוֹנִי מֵאֲחוֹרֵי גַּבָּהּ.
  7. הַמּוֹרָה דַּפְנָה לִימְּדָה חֶשְׁבּוֹן, וְלֹא אָמְרָה אֲפִילּוּ מִילָּה אַחַת עַל יוֹם הַהוּלֶּדֶת.
  8. בַּהַפְסָקָה יָשְׁבָה שִׁירָן עַל סַפְסָל בֶּחָצֵר, וְעֵינֶיהָ הִתְמַלְּאוּ דְּמָעוֹת.
  9. "הַמּוֹרָה פָּשׁוּט שָׁכְחָה אוֹתִי," לָחֲשָׁה לְגִלְעָד.
  10. "גַּם אֲנִי חוֹשֵׁב כָּךְ," נֶאֱנַח גִּלְעָד. "וּמוּזָר שֶׁאַף יֶלֶד לֹא אָמַר לָךְ מַזָּל טוֹב."
  11. "אוּלַי כְּדַאי לְהַזְכִּיר לָהּ?" הִצִּיעַ. שִׁירָן הֵנִידָה בְּרֹאשָׁהּ: "לֹא, אֲנִי מִתְבַּיֶּישֶׁת."
  12. לִפְנֵי הַשִּׁיעוּר הָאַחֲרוֹן נִיגְּשָׁה הַמּוֹרָה אֶל שִׁירָן וְהוֹשִׁיטָה לָהּ מַעֲטָפָה.
  13. "שִׁירָן, בְּבַקָּשָׁה קְחִי אֶת הַמַּעֲטָפָה לַמַּזְכִּירוּת. גִּלְעָד יְלַוֶּוה אוֹתָךְ," אָמְרָה.
  14. הֵם הָלְכוּ לְאַט בַּמִּסְדְּרוֹן. "אֲפִילּוּ עַכְשָׁיו הִיא לֹא נִזְכְּרָה," מִלְמְלָה שִׁירָן.
  15. כְּשֶׁחָזְרוּ, הַדֶּלֶת הָיְתָה סְגוּרָה, וְהַכִּיתָּה הָיְתָה חֲשׁוּכָה וְדוֹמֶמֶת.
  16. "לְאָן נֶעֶלְמוּ כּוּלָּם?" תָּמַהּ גִּלְעָד, וּפָתַח אֶת הַדֶּלֶת בִּזְהִירוּת.
  17. פִּתְאוֹם נִדְלַק הָאוֹר. "הַפְתָּעָה!" צָעֲקוּ כָּל הַיְּלָדִים, וּפָרְצוּ בְּשִׁיר.
  18. הַכִּיתָּה הָיְתָה מְקוּשֶּׁטֶת בְּבָלוֹנִים, וְעַל הַשּׁוּלְחָן חִיכְּתָה עוּגָה גְּדוֹלָה.
  19. עַל הַלּוּחַ נִתְלָה דַּף צִבְעוֹנִי, וְעָלָיו בְּרָכוֹת וְצִיּוּרִים מִכָּל יַלְדֵי הַכִּיתָּה.
  20. "עַכְשָׁיו אֲנִי מְבִינָה עַל מָה הֵם הִתְלַחֲשׁוּ בַּבּוֹקֶר!" צָחֲקָה שִׁירָן.
  21. הַמּוֹרָה הִנִּיחָה כֶּתֶר זָהוֹב עַל רֹאשָׁהּ: "אֶת הַכֶּתֶר הֵכַנּוּ כְּבָר לִפְנֵי כַּמָּה יָמִים!"
  22. שִׁירָן חִיְּיכָה מֵאוֹזֶן לְאוֹזֶן. "זֶה יוֹם הַהוּלֶּדֶת הֲכִי יָפֶה שֶׁהָיָה לִי," אָמְרָה.

עִבְרוּ לָעַמּוּד הַבָּא.

כָּעֵת, עֲנוּ עַל הַשְּׁאֵלוֹת שֶׁלִּפְנֵיכֶם.
הַקִּיפוּ אֶת הַתְּשׁוּבָה הַנְּכוֹנָה בְּכָל שְׁאֵלָה.

1. בְּשׁוּרוֹת 2–3 מְסוּפָּר מָה קוֹרֶה בַּכִּיתָּה כְּשֶׁיֶּלֶד חוֹגֵג יוֹם הוּלֶּדֶת.
מַדּוּעַ מְסוּפָּר עַל כָּךְ בִּתְחִילַּת הַסִּיפּוּר?
  1. א.כְּדֵי לְשַׁכְנֵעַ מוֹרוֹת אֲחֵרוֹת לַחְגּוֹג יְמֵי הוּלֶּדֶת בַּכִּיתָּה
