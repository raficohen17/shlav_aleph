חֵלֶק רִאשׁוֹן: קִטְעֵי קְרִיאָה וּשְׁאֵלוֹת

קִרְאוּ אֶת קֶטַע הַמֵּידָע שֶׁלִּפְנֵיכֶם וַעֲנוּ עַל הַשְּׁאֵלוֹת שֶׁאַחֲרָיו.

קוּרֵי הָעַכָּבִישׁ
  1. בְּפִינַּת הַחֶדֶר אוֹ בֵּין עַנְפֵי הָעֵץ אֶפְשָׁר לִרְאוֹת לִפְעָמִים רֶשֶׁת דַּקָּה וַעֲדִינָה.
  2. אֶת הָרֶשֶׁת הַזֹּאת בָּנָה עַכָּבִישׁ, וְקוֹרְאִים לָהּ "קוּרֵי עַכָּבִישׁ".
  3. הָעַכָּבִישׁ מְיַיצֵּר בְּתוֹךְ גּוּפוֹ חוּט מֶשִׁי דַּק, וּמִמֶּנּוּ הוּא טוֹוֶה אֶת הָרֶשֶׁת.
  4. הַחוּט נִרְאֶה חַלָּשׁ, אֲבָל לְמַעֲשֶׂה הוּא חָזָק מְאוֹד. חוּט מֶשִׁי שֶׁל עַכָּבִישׁ
  5. חָזָק אֲפִילּוּ יוֹתֵר מֵחוּט פְּלָדָה בְּאוֹתוֹ עוֹבִי. פְּלָדָה הִיא מַתֶּכֶת קָשָׁה וַחֲזָקָה!
  6. לָמָּה הָעַכָּבִישׁ צָרִיךְ רֶשֶׁת? הָרֶשֶׁת הִיא הַמַּלְכּוֹדֶת שֶׁלּוֹ. חֵלֶק מֵהַחוּטִים
  7. בָּרֶשֶׁת דְּבִיקִים כְּמוֹ דֶּבֶק. זְבוּב אוֹ יַתּוּשׁ שֶׁעָף אֶל הָרֶשֶׁת נִדְבָּק אֵלֶיהָ
  8. וְאֵינוֹ מַצְלִיחַ לְהִשְׁתַּחְרֵר. הָעַכָּבִישׁ מַרְגִּישׁ שֶׁהָרֶשֶׁת רוֹעֶדֶת, וּמְמַהֵר אֶל הַטֶּרֶף.
  9. וְאֵיךְ הָעַכָּבִישׁ עַצְמוֹ אֵינוֹ נִדְבָּק לָרֶשֶׁת שֶׁבָּנָה? הַתְּשׁוּבָה מַפְתִּיעָה:
  10. לֹא כָּל הַחוּטִים בָּרֶשֶׁת דְּבִיקִים. הַחוּטִים שֶׁיּוֹצְאִים מִן הַמֶּרְכָּז הַחוּצָה
  11. יְבֵשִׁים וַחֲלָקִים. הָעַכָּבִישׁ יוֹדֵעַ הֵיטֵב אֵילוּ חוּטִים יְבֵשִׁים, וְהוֹלֵךְ רַק עֲלֵיהֶם.
  12. הָרוּחַ וְהַגֶּשֶׁם פּוֹגְעִים בָּרֶשֶׁת, וְהִיא נִקְרַעַת לְעִיתִּים קְרוֹבוֹת.
  13. הָעַכָּבִישׁ אֵינוֹ מִתְיָיאֵשׁ: כִּמְעַט בְּכָל יוֹם הוּא בּוֹנֶה רֶשֶׁת חֲדָשָׁה.
  14. לִפְנֵי כֵן הוּא אוֹכֵל אֶת הָרֶשֶׁת הַיְּשָׁנָה, וְכָךְ הַמֶּשִׁי הַיָּקָר אֵינוֹ מִתְבַּזְבֵּז.
  15. יֵשׁ אֲנָשִׁים שֶׁפּוֹחֲדִים מֵעַכְּבִישִׁים, אֲבָל רוֹב הָעַכְּבִישִׁים אֵינָם מְסוּכָּנִים לִבְנֵי אָדָם.
  16. לְהֶפֶךְ: הֵם מוֹעִילִים לָנוּ, כִּי הֵם טוֹרְפִים זְבוּבִים וְיַתּוּשִׁים שֶׁמַּטְרִידִים אוֹתָנוּ.

עִבְרוּ לָעַמּוּד הַבָּא.

כָּעֵת, עֲנוּ עַל הַשְּׁאֵלוֹת שֶׁלִּפְנֵיכֶם.
הַקִּיפוּ אֶת הַתְּשׁוּבָה הַנְּכוֹנָה בְּכָל שְׁאֵלָה.

1. מַדּוּעַ הָעַכָּבִישׁ אֵינוֹ נִדְבָּק לָרֶשֶׁת שֶׁלּוֹ?
  1. א.כִּי הָרַגְלַיִים שֶׁלּוֹ מְכוּסּוֹת בְּשֶׁמֶן
  2. ב.כִּי הוּא אוֹכֵל אֶת כָּל הַחוּטִים הַדְּבִיקִים
  3. ג.כִּי הוּא הוֹלֵךְ רַק עַל הַחוּטִים הַיְּבֵשִׁים
  4. ד.כִּי הַחוּט שֶׁלּוֹ חָזָק יוֹתֵר מִפְּלָדָה
2. מַה פֵּירוּשׁ הַמִּילָּה "מוֹעִילִים" (שׁוּרָה 16)?
  1. א.מַפְרִיעִים
  2. ב.עוֹזְרִים
  3. ג.מְסוּכָּנִים
  4. ד.מַפְחִידִים
3. בְּשׁוּרָה 5 כָּתוּב שֶׁחוּט הַמֶּשִׁי חָזָק אֲפִילּוּ יוֹתֵר מֵחוּט פְּלָדָה.
מַדּוּעַ כָּתוּב "אֲפִילּוּ"?
  1. א.כִּי מַפְתִּיעַ שֶׁחוּט דַּק כָּזֶה חָזָק יוֹתֵר מִמַּתֶּכֶת חֲזָקָה
  2. ב.כִּי גַּם הָעַכָּבִישׁ עַצְמוֹ חָזָק מְאוֹד, יוֹתֵר מִכָּל חֶרֶק אַחֵר
  3. ג.כִּי הָעַכָּבִישׁ בּוֹנֶה אֶת הָרֶשֶׁת שֶׁלּוֹ מֵחוּטֵי פְּלָדָה דַּקִּים
  4. ד.כִּי חוּט פְּלָדָה הוּא חוּט חַלָּשׁ מְאוֹד
4. מָה אֶפְשָׁר לִלְמוֹד מִשּׁוּרוֹת 12–14?
  1. א.שֶׁהָעַכָּבִישׁ בּוֹנֶה רַק רֶשֶׁת אַחַת בְּכָל חַיָּיו
  2. ב.שֶׁהָעַכָּבִישׁ פּוֹחֵד מְאוֹד מֵהָרוּחַ וּמֵהַגֶּשֶׁם
  3. ג.שֶׁרֶשֶׁת יְשָׁנָה דְּבִיקָה יוֹתֵר מֵרֶשֶׁת חֲדָשָׁה
  4. ד.שֶׁהָעַכָּבִישׁ חָרוּץ, וְאֵינוֹ מְבַזְבֵּז אֶת הַמֶּשִׁי שֶׁלּוֹ
5. אֵיזֶה מִשְׁפָּט אִי אֶפְשָׁר לוֹמַר עַל פִּי הַקֶּטַע?
  1. א.חֵלֶק מֵהַחוּטִים בָּרֶשֶׁת אֵינָם דְּבִיקִים.
  2. ב.הָעַכָּבִישׁ מַרְגִּישׁ כְּשֶׁזְּבוּב נִלְכָּד בָּרֶשֶׁת.
  3. ג.רוֹב הָעַכְּבִישִׁים מְסוּכָּנִים לִבְנֵי אָדָם.
  4. ד.עַכְּבִישִׁים טוֹרְפִים זְבוּבִים וְיַתּוּשִׁים.

עִבְרוּ לָעַמּוּד הַבָּא.

קִרְאוּ אֶת הַסִּיפּוּר שֶׁלִּפְנֵיכֶם וַעֲנוּ עַל הַשְּׁאֵלוֹת שֶׁאַחֲרָיו.

מַה מִּסְתַּתֵּר בַּמַּחְסָן?
  1. מָאיָה, יַלְדָּה בַּת שְׁמוֹנֶה, וְאָחִיהָ הַקָּטָן אֵיתָן גָּרִים בְּבַיִת עִם חָצֵר גְּדוֹלָה.
  2. בִּקְצֵה הֶחָצֵר עוֹמֵד מַחְסָן יָשָׁן. בְּכָל עֶרֶב מָאיָה מַנִּיחָה לְיַד הַדֶּלֶת שֶׁלּוֹ
  3. קַעֲרַת חָלָב בִּשְׁבִיל חֲתוּלָה אֲפוֹרָה שֶׁמְּבַקֶּרֶת בֶּחָצֵר.
  4. כְּבָר שָׁבוּעַ שֶׁהַחֲתוּלָה לֹא נִרְאֲתָה בֶּחָצֵר, וּמָאיָה דָּאֲגָה לָהּ מְאוֹד.
  5. "מוּזָר," אָמְרָה לְאֵיתָן, "הַחֲתוּלָה נֶעֶלְמָה, וּבְכָל זֹאת הַקְּעָרָה רֵיקָה בְּכָל בּוֹקֶר."
  6. לַיְלָה אֶחָד הִתְעוֹרֵר אֵיתָן וְשָׁמַע רְעָשִׁים מוּזָרִים מִכִּיוּוּן הַמַּחְסָן.
  7. הוּא רָץ אֶל הַמִּיטָּה שֶׁל מָאיָה וְהֵעִיר אוֹתָהּ. "מָאיָה, תַּקְשִׁיבִי!" לָחַשׁ.
  8. מָאיָה הִקְשִׁיבָה. מִן הַמַּחְסָן נִשְׁמְעוּ רִשְׁרוּשִׁים, חֲרִיקוֹת וִילָלוֹת דַּקּוֹת.
  9. "יֵשׁ שָׁם גַּנָּב!" אָמַר אֵיתָן, וְקוֹלוֹ רָעַד. "הוּא בָּא לִגְנוֹב אֶת הָאוֹפַנַּיִים שֶׁלָּנוּ!"
  10. גַּם מָאיָה נִבְהֲלָה, אֲבָל הִיא לֹא רָצְתָה שֶׁאֵיתָן יִפְחַד עוֹד יוֹתֵר.
  11. "גַּנָּב לֹא מְיַילֵּל," אָמְרָה בְּקוֹל רָגוּעַ. "בַּבּוֹקֶר נִבְדּוֹק יַחַד מַה יֵּשׁ שָׁם."
  12. בַּבּוֹקֶר לָקְחָה מָאיָה פָּנָס, וְאֵיתָן הֶחְזִיק מַטְאֲטֵא גָּדוֹל – לְיֶתֶר בִּיטָּחוֹן.
  13. הֵם הָלְכוּ בְּשֶׁקֶט אֶל הַמַּחְסָן. הַקְּעָרָה שֶׁלְּיַד הַדֶּלֶת הָיְתָה שׁוּב רֵיקָה.
  14. מָאיָה דָּחֲפָה לְאַט אֶת הַדֶּלֶת, שֶׁהָיְתָה פְּתוּחָה מְעַט, וְהֵאִירָה בַּפָּנָס פְּנִימָה.
  15. בְּתוֹךְ אַרְגַּז קַרְטוֹן יָשָׁן, עַל שְׂמִיכָה קְרוּעָה, שָׁכְבָה הַחֲתוּלָה הָאֲפוֹרָה.
  16. לְיָדָהּ הִצְטוֹפְפוּ אַרְבָּעָה גּוּרִים זְעִירִים, וְעֵינֵיהֶם עֲדַיִין עֲצוּמוֹת.
  17. הַגּוּרִים יִילְּלוּ בְּקוֹל דַּק וְנִצְמְדוּ אֶל אִימָּא שֶׁלָּהֶם.
  18. "אָז זֶה הַגַּנָּב שֶׁלְּךָ!" צָחֲקָה מָאיָה. אֵיתָן הִנִּיחַ אֶת הַמַּטְאֲטֵא וְחִיֵּיךְ בִּמְבוּכָה.
  19. "עַכְשָׁיו אֲנִי מְבִינָה לְאָן נֶעֱלַם הֶחָלָב," אָמְרָה מָאיָה. "אִימָּא חֲתוּלָה צְרִיכָה כּוֹחַ."
  20. "מֵהַיּוֹם אֲנִי אֲמַלֵּא אֶת הַקְּעָרָה," אָמַר אֵיתָן, "וְאָבִיא לָהֶם גַּם שְׂמִיכָה חֲדָשָׁה."
  21. הוּא הִתְכּוֹפֵף וְלָחַשׁ לַגּוּרִים: "אַל תִּדְאֲגוּ, אֲנִי אֶשְׁמוֹר עֲלֵיכֶם מִפְּנֵי גַּנָּבִים."
  22. בְּאוֹתוֹ לַיְלָה שׁוּב נִשְׁמְעוּ יְלָלוֹת מִן הַמַּחְסָן, אֲבָל הַפַּעַם אֵיתָן נִרְדַּם בְּחִיּוּךְ.

עִבְרוּ לָעַמּוּד הַבָּא.

כָּעֵת, עֲנוּ עַל הַשְּׁאֵלוֹת שֶׁלִּפְנֵיכֶם.
הַקִּיפוּ אֶת הַתְּשׁוּבָה הַנְּכוֹנָה בְּכָל שְׁאֵלָה.

1. בְּשׁוּרָה 12 כָּתוּב שֶׁאֵיתָן הֶחְזִיק מַטְאֲטֵא גָּדוֹל – לְיֶתֶר בִּיטָּחוֹן.
הַמַּטָּרָה שֶׁל הַמִּשְׁפָּט הַזֶּה הִיא –
  1. א.לְהַרְאוֹת שֶׁאֵיתָן עֲדַיִין פָּחַד מִמָּה שֶׁיֵּשׁ בַּמַּחְסָן
  2. ב.לְסַפֵּר שֶׁאֵיתָן רָצָה לְנַקּוֹת אֶת הַמַּחְסָן
  3. ג.לְהַסְבִּיר שֶׁהַמַּטְאֲטֵא שַׁיָּיךְ לְאֵיתָן
  4. ד.לְהַדְגִּישׁ שֶׁאֵיתָן חָזָק יוֹתֵר מִמָּאיָה
2. מַה פֵּירוּשׁ הַמִּילָּה "הִצְטוֹפְפוּ" (שׁוּרָה 16)?
  1. א.רָבוּ זֶה עִם זֶה
  2. ב.בָּרְחוּ לְכָל הַכִּיוּוּנִים
