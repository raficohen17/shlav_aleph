חֵלֶק רִאשׁוֹן: קִטְעֵי קְרִיאָה וּשְׁאֵלוֹת

קִרְאוּ אֶת הַקֶּטַע שֶׁלִּפְנֵיכֶם וַעֲנוּ עַל הַשְּׁאֵלוֹת שֶׁאַחֲרָיו.

אֵיךְ שׁוֹמְרִים עַל שִׁינַּיִים בְּרִיאוֹת?
  1. בְּתוֹךְ הַפֶּה שֶׁלָּנוּ חַיִּים חַיְידַּקִּים זְעִירִים, שֶׁאִי אֶפְשָׁר לִרְאוֹת בָּעַיִן.
  2. הַחַיְידַּקִּים הָאֵלֶּה אוֹהֲבִים מְאוֹד סוּכָּר. כְּשֶׁאֲנַחְנוּ אוֹכְלִים מַמְתָּקִים, הֵם אוֹכְלִים
  3. אֶת הַסּוּכָּר שֶׁנִּשְׁאָר עַל הַשִּׁינַּיִים, וְיוֹצְרִים מִמֶּנּוּ חוּמְצָה.
  4. הַחוּמְצָה פּוֹגַעַת בַּשֵּׁן לְאַט לְאַט, עַד שֶׁנּוֹצָר בָּהּ חוֹר. חוֹר כָּזֶה עָלוּל לִכְאוֹב.
  5. אֵיךְ אֶפְשָׁר לִמְנוֹעַ אֶת זֶה? הִנֵּה שָׁלוֹשׁ הַמְלָצוֹת שֶׁל רוֹפְאֵי שִׁינַּיִים:
  6. הַמְלָצָה 1: צַחְצְחוּ שִׁינַּיִים פַּעֲמַיִים בְּיוֹם – בַּבּוֹקֶר, וּבָעֶרֶב לִפְנֵי הַשֵּׁינָה.
  7. כָּל צִחְצוּחַ צָרִיךְ לְהִימָּשֵׁךְ שְׁתֵּי דַּקּוֹת לְפָחוֹת. צִחְצוּחַ חָטוּף שֶׁל כַּמָּה שְׁנִיּוֹת
  8. אֵינוֹ מְנַקֶּה אֶת כָּל הַשִּׁינַּיִים, וְהַחַיְידַּקִּים נִשְׁאָרִים בַּפֶּה.
  9. הַצִּחְצוּחַ בָּעֶרֶב חָשׁוּב בִּמְיוּחָד: בַּלַּיְלָה הַחַיְידַּקִּים פּוֹעֲלִים שָׁעוֹת רַבּוֹת.
  10. הַמְלָצָה 2: אִכְלוּ מְעַט מַמְתָּקִים, וְאַל תְּנַשְׁנְשׁוּ אוֹתָם בְּמֶשֶׁךְ כָּל הַיּוֹם.
  11. אִם אֲכַלְתֶּם מַשֶּׁהוּ מָתוֹק וְאֵין לָכֶם מִבְרֶשֶׁת, שִׁטְפוּ אֶת הַפֶּה בְּמַיִם.
  12. שִׂימוּ לֵב: אֲפִילּוּ בְּמִיץ פֵּירוֹת וּבְפֵירוֹת יְבֵשִׁים יֵשׁ הַרְבֵּה סוּכָּר!
  13. הַמְלָצָה 3: בַּקְּרוּ אֵצֶל רוֹפֵא שִׁינַּיִים פַּעֲמַיִים בְּשָׁנָה, גַּם כְּשֶׁשּׁוּם דָּבָר אֵינוֹ כּוֹאֵב.
  14. הָרוֹפֵא יָכוֹל לְגַלּוֹת חוֹר קָטָן וּלְטַפֵּל בּוֹ לִפְנֵי שֶׁהוּא גָּדֵל וּמַתְחִיל לִכְאוֹב.
  15. הַשִּׁינַּיִים הַקְּבוּעוֹת שֶׁצּוֹמְחוֹת לָכֶם עַכְשָׁיו צְרִיכוֹת לְהַחְזִיק מַעֲמָד כָּל הַחַיִּים.
  16. שִׁמְרוּ עֲלֵיהֶן הֵיטֵב – וְתוּכְלוּ לְחַיֵּיךְ חִיּוּךְ בָּרִיא וְיָפֶה גַּם בְּעוֹד שָׁנִים רַבּוֹת.

עִבְרוּ לָעַמּוּד הַבָּא.

כָּעֵת, עֲנוּ עַל הַשְּׁאֵלוֹת שֶׁלִּפְנֵיכֶם.
הַקִּיפוּ אֶת הַתְּשׁוּבָה הַנְּכוֹנָה בְּכָל שְׁאֵלָה.

1. לְפִי הַקֶּטַע, מָה גּוֹרֵם לְחוֹר בַּשֵּׁן?
  1. א.הַמִּבְרֶשֶׁת, שֶׁמְּשַׁפְשֶׁפֶת אֶת הַשֵּׁן חָזָק מִדַּי
  2. ב.הַמַּיִם, שֶׁשּׁוֹטְפִים אֶת הַפֶּה אַחֲרֵי הָאוֹכֶל
  3. ג.הַחוּמְצָה, שֶׁהַחַיְידַּקִּים יוֹצְרִים מִן הַסּוּכָּר
  4. ד.הַסּוּכָּר, שֶׁהַחַיְידַּקִּים יוֹצְרִים מִן הַחוּמְצָה
2. מָה פֵּירוּשׁ הַמִּילָה "חָטוּף" (שׁוּרָה 7)?
  1. א.קָצָר וּמָהִיר
  2. ב.חָזָק וְכוֹאֵב
  3. ג.אִיטִּי וִיסוֹדִי
  4. ד.מְאוּחָר בַּלַּיְלָה
3. בְּשׁוּרָה 12 כָּתוּב: "אֲפִילּוּ בְּמִיץ פֵּירוֹת וּבְפֵירוֹת יְבֵשִׁים יֵשׁ הַרְבֵּה סוּכָּר".
מַדּוּעַ כָּתוּב "אֲפִילּוּ"?
  1. א.כִּי מִיץ פֵּירוֹת טָעִים יוֹתֵר מִמַּמְתָּקִים
  2. ב.כִּי בְּדֶרֶךְ כְּלָל לֹא חוֹשְׁבִים שֶׁהֵם מַמְתָּקִים, וְלָכֵן זֶה מַפְתִּיעַ
  3. ג.כִּי רוֹפְאֵי הַשִּׁינַּיִים מַמְלִיצִים לִשְׁתּוֹת הַרְבֵּה מִיץ
  4. ד.כִּי הַחַיְידַּקִּים אֵינָם אוֹהֲבִים פֵּירוֹת
4. נִיר אָכַל עוּגַת שׁוֹקוֹלָד בְּטִיּוּל, וְאֵין לוֹ מִבְרֶשֶׁת שִׁינַּיִים.
לְפִי הַקֶּטַע, מָה כְּדַאי לְנִיר לַעֲשׂוֹת?
  1. א.לֶאֱכוֹל פֵּירוֹת יְבֵשִׁים
  2. ב.לְחַכּוֹת לַבִּיקּוּר אֵצֶל רוֹפֵא הַשִּׁינַּיִים
  3. ג.לִשְׁתּוֹת מִיץ פֵּירוֹת
  4. ד.לִשְׁטוֹף אֶת הַפֶּה בְּמַיִם
5. הַשְׁלִימוּ אֶת הַמִּשְׁפָּט לְפִי הַקֶּטַע. כִּתְבוּ מִילָּה אַחַת עַל הַקַּו.
כְּדַאי לְבַקֵּר אֵצֶל רוֹפֵא שִׁינַּיִים בְּשָׁנָה.

עִבְרוּ לָעַמּוּד הַבָּא.

קִרְאוּ אֶת הַסִּיפּוּר שֶׁלִּפְנֵיכֶם וַעֲנוּ עַל הַשְּׁאֵלוֹת שֶׁאַחֲרָיו.

הַיּוֹם הָרִאשׁוֹן שֶׁל רוֹתֶם
  1. בְּאֶמְצַע הַשָּׁנָה עָבְרָה רוֹתֶם עִם מִשְׁפַּחְתָּהּ לְעִיר אַחֶרֶת, וְהִגִּיעָה לְבֵית סֵפֶר חָדָשׁ.
  2. בַּבּוֹקֶר הָרִאשׁוֹן הִיא עָמְדָה לְיַד הַשַּׁעַר, וְהַלֵּב שֶׁלָּהּ דָּפַק בְּחוֹזְקָה.
  3. "אֲנִי לֹא מַכִּירָה כָּאן אַף אֶחָד," לָחֲשָׁה לְאִימָּא. אִימָּא חִיבְּקָה אוֹתָהּ וְאָמְרָה:
  4. "גַּם הַיְּלָדִים עֲדַיִין לֹא מַכִּירִים אוֹתָךְ. תְּנִי לָהֶם קְצָת זְמַן."
  5. הַמּוֹרָה אוֹרִית הוֹשִׁיבָה אֶת רוֹתֶם לְיַד יַלְדָּה שְׁקֵטָה בְּשֵׁם נֶטַע.
  6. נֶטַע הִנְהֲנָה לְעֶבְרָהּ, אֲבָל לֹא אָמְרָה מִילָּה. רוֹתֶם הִרְגִּישָׁה בּוֹדֵדָה עוֹד יוֹתֵר.
  7. בַּהַפְסָקָה יָצְאוּ כָּל הַיְּלָדִים לְשַׂחֵק בֶּחָצֵר, וְרוֹתֶם יָשְׁבָה לְבַדָּהּ עַל סַפְסָל.
  8. כְּשֶׁהַפַּעֲמוֹן צִלְצֵל, הִיא מִיהֲרָה לַחֲזוֹר, פָּתְחָה דֶּלֶת וְהִתְיַישְּׁבָה בַּכִּיסֵּא הָרִאשׁוֹן.
  9. פִּתְאוֹם שָׂמָה לֵב שֶׁכָּל הַיְּלָדִים סְבִיבָהּ גְּדוֹלִים מִמֶּנָּה, וְכוּלָּם מִסְתַּכְּלִים עָלֶיהָ.
  10. "נִרְאֶה לִי שֶׁהִתְבַּלְבַּלְתְּ," חִיֵּיךְ הַמּוֹרֶה. "זֹאת כִּיתָּה ו'. כִּיתָּה ב' בַּדֶּלֶת הַבָּאָה."
  11. הַיְּלָדִים פָּרְצוּ בִּצְחוֹק. הַפָּנִים שֶׁל רוֹתֶם הֶאֱדִימוּ, וְהִיא רָצְתָה לְהֵיעָלֵם.
  12. הִיא חָזְרָה לַכִּיתָּה שֶׁלָּהּ בְּרֹאשׁ מוּרְכָּן, הוֹצִיאָה דַּף וְהִתְחִילָה לְצַיֵּיר.
  13. הִיא צִיְּירָה יַלְדָּה קְטַנָּה שֶׁיּוֹשֶׁבֶת בֵּין עֲנָקִים, וְהָעֲנָקִים צוֹחֲקִים.
  14. נֶטַע הֵצִיצָה בַּדַּף. "זֹאת אַתְּ, בְּכִיתָּה ו'?" שָׁאֲלָה בְּשֶׁקֶט. רוֹתֶם הִנְהֲנָה.
  15. "בַּשָּׁבוּעַ הָרִאשׁוֹן שֶׁלִּי כָּאן נִכְנַסְתִּי בְּטָעוּת לַחֲדַר הַמּוֹרִים," סִיפְּרָה נֶטַע.
  16. רוֹתֶם הֵרִימָה אֶת הָרֹאשׁ. "בֶּאֱמֶת? גַּם אַתְּ הָיִית חֲדָשָׁה?"
  17. "בַּשָּׁנָה שֶׁעָבְרָה," עָנְתָה נֶטַע. הִיא לָקְחָה עִיפָּרוֹן וְהוֹסִיפָה עֲנָק עִם מִשְׁקָפַיִים.
  18. "זֶה הַמּוֹרֶה," אָמְרָה. שְׁתֵּיהֶן הִתְפַּקְּעוּ מִצְּחוֹק, עַד שֶׁהַמּוֹרָה אוֹרִית בִּיקְּשָׁה שֶׁקֶט.
  19. עַד סוֹף הַיּוֹם הֵן צִיְּירוּ יַחַד דַּף שָׁלֵם שֶׁל עֲנָקִים מַצְחִיקִים.
  20. כְּשֶׁאִימָּא בָּאָה לֶאֱסוֹף אוֹתָהּ, רוֹתֶם רָצָה אֵלֶיהָ עִם הַצִּיּוּר בַּיָּד.
  21. "אֵיךְ הָיָה?" שָׁאֲלָה אִימָּא. "בַּהַתְחָלָה הָיָה נוֹרָא," עָנְתָה רוֹתֶם,
  22. "אֲבָל מָחָר אֲנִי בָּאָה מוּקְדָּם. נֶטַע וַאֲנִי צְרִיכוֹת לְסַיֵּים אֶת הַצִּיּוּר."

עִבְרוּ לָעַמּוּד הַבָּא.

כָּעֵת, עֲנוּ עַל הַשְּׁאֵלוֹת שֶׁלִּפְנֵיכֶם.
הַקִּיפוּ אֶת הַתְּשׁוּבָה הַנְּכוֹנָה בְּכָל שְׁאֵלָה.

1. מַדּוּעַ צָחֲקוּ הַיְּלָדִים בְּכִיתָּה ו' (שׁוּרָה 11)?
  1. א.כִּי רוֹתֶם צִיְּירָה צִיּוּר מַצְחִיק
  2. ב.כִּי רוֹתֶם נִכְנְסָה בְּטָעוּת לַכִּיתָּה שֶׁלָּהֶם
  3. ג.כִּי הַמּוֹרֶה סִיפֵּר לָהֶם בְּדִיחָה
  4. ד.כִּי רוֹתֶם נִכְנְסָה בְּטָעוּת לַחֲדַר הַמּוֹרִים
2. בְּשׁוּרָה 17 כָּתוּב: "הִיא לָקְחָה עִיפָּרוֹן". מִי זֹאת "הִיא"?
  1. א.רוֹתֶם
  2. ב.הַמּוֹרָה אוֹרִית
  3. ג.אִימָּא
  4. ד.נֶטַע
3. בְּשׁוּרָה 15 נֶטַע מְסַפֶּרֶת שֶׁנִּכְנְסָה בְּטָעוּת לַחֲדַר הַמּוֹרִים.
