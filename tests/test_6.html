חֵלֶק רִאשׁוֹן: קִטְעֵי קְרִיאָה וּשְׁאֵלוֹת

קִרְאוּ אֶת קֶטַע הַמֵּידָע שֶׁלִּפְנֵיכֶם וַעֲנוּ עַל הַשְּׁאֵלוֹת שֶׁאַחֲרָיו.

הַדְּבוֹרִים
  1. בְּכָל כַּוֶּרֶת חַיּוֹת יַחַד אַלְפֵי דְּבוֹרִים, וּלְכָל דְּבוֹרָה יֵשׁ תַּפְקִיד.
  2. הַמַּלְכָּה הִיא הַדְּבוֹרָה הַגְּדוֹלָה בְּיוֹתֵר, וְרַק הִיא מְטִילָה בֵּיצִים.
  3. דְּבוֹרִים צְעִירוֹת מְנַקּוֹת אֶת הַכַּוֶּרֶת וּמַאֲכִילוֹת אֶת הַזְּחָלִים הַקְּטַנִּים.
  4. דְּבוֹרִים אֲחֵרוֹת עוֹמְדוֹת בַּפֶּתַח וְשׁוֹמְרוֹת שֶׁאוֹיֵב לֹא יִיכָּנֵס פְּנִימָה.
  5. הַדְּבוֹרִים הַמְּבוּגָּרוֹת יוֹצְאוֹת הַחוּצָה וְאוֹסְפוֹת צוּף מָתוֹק מִן הַפְּרָחִים.
  6. מִן הַצּוּף הַזֶּה הַדְּבוֹרִים מְכִינוֹת בַּכַּוֶּרֶת אֶת הַדְּבַשׁ.
  7. כְּשֶׁדְּבוֹרָה מוֹצֵאת שָׂדֶה מָלֵא פְּרָחִים, הִיא מְמַהֶרֶת לַחֲזוֹר אֶל הַכַּוֶּרֶת.
  8. אֵיךְ הִיא מְסַפֶּרֶת לַחֲבֵרוֹתֶיהָ הֵיכָן הַשָּׂדֶה? הֲרֵי דְּבוֹרִים אֵינָן מְדַבְּרוֹת!
  9. הַדְּבוֹרָה רוֹקֶדֶת לִפְנֵיהֶן רִיקּוּד מְיוּחָד, וְהֵן עוֹקְבוֹת אַחֲרֶיהָ.
  10. אִם הַפְּרָחִים קְרוֹבִים, הִיא רוֹקֶדֶת בְּמַעְגָּל קָטָן.
  11. אִם הַפְּרָחִים רְחוֹקִים, הִיא מְנַעְנַעַת אֶת גּוּפָהּ מִצַּד לְצַד.
  12. כְּכָל שֶׁהַפְּרָחִים רְחוֹקִים יוֹתֵר, הָרִיקּוּד נִמְשָׁךְ זְמַן רַב יוֹתֵר.
  13. הָרִיקּוּד מַרְאֶה אֲפִילּוּ לְאֵיזֶה כִּיוּוּן צָרִיךְ לָעוּף כְּדֵי לְהַגִּיעַ אֲלֵיהֶם.
  14. אֲנָשִׁים רַבִּים פּוֹחֲדִים מִדְּבוֹרִים, אֲבָל דְּבוֹרָה עוֹקֶצֶת רַק כְּשֶׁהִיא בְּסַכָּנָה.
  15. לְמַעֲשֶׂה, דְּבוֹרַת הַדְּבַשׁ יְכוֹלָה לַעֲקוֹץ אָדָם רַק פַּעַם אַחַת.
  16. הָעוֹקֶץ שֶׁלָּהּ נִתְקָע בָּעוֹר, וּזְמַן קָצָר אַחֲרֵי הָעֲקִיצָה הִיא מֵתָה.
  17. לָכֵן, אִם לֹא תַּפְרִיעוּ לַדְּבוֹרָה, גַּם הִיא לֹא תַּפְרִיעַ לָכֶם.

עִבְרוּ לָעַמּוּד הַבָּא.

כָּעֵת, עֲנוּ עַל הַשְּׁאֵלוֹת שֶׁלִּפְנֵיכֶם.
הַקִּיפוּ אֶת הַתְּשׁוּבָה הַנְּכוֹנָה בְּכָל שְׁאֵלָה.

1. מַדּוּעַ דְּבוֹרָה שֶׁמָּצְאָה שָׂדֶה מָלֵא פְּרָחִים רוֹקֶדֶת לִפְנֵי חֲבֵרוֹתֶיהָ?
  1. א.כְּדֵי לְשַׂמֵּחַ אֶת הַמַּלְכָּה
  2. ב.כִּי הִיא רוֹצָה לְהַפְחִיד אוֹיֵב שֶׁנִּכְנַס לַכַּוֶּרֶת
  3. ג.כְּדֵי לְסַפֵּר לָהֶן הֵיכָן נִמְצָאִים הַפְּרָחִים
  4. ד.כִּי כָּךְ הִיא מְכִינָה דְּבַשׁ מִן הַצּוּף
2. בְּשׁוּרָה 9 כָּתוּב: "הַדְּבוֹרָה רוֹקֶדֶת לִפְנֵיהֶן רִיקּוּד מְיוּחָד, וְהֵן עוֹקְבוֹת אַחֲרֶיהָ".
לְמִי אוֹ לְמָה מִתְיַיחֶסֶת הַמִּילָה "הֵן"?
  1. א.לַדְּבוֹרִים שֶׁרוֹאוֹת אֶת הָרִיקּוּד
  2. ב.לַבֵּיצִים שֶׁהַמַּלְכָּה מְטִילָה
  3. ג.לַדְּבוֹרִים שֶׁנִּשְׁאֲרוּ בִּשְׂדֵה הַפְּרָחִים
  4. ד.לָרַגְלַיִים שֶׁל הַדְּבוֹרָה הָרוֹקֶדֶת
3. מָה אֶפְשָׁר לִלְמוֹד מִשּׁוּרוֹת 2–5?
  1. א.שֶׁכָּל הַדְּבוֹרִים בַּכַּוֶּרֶת מְטִילוֹת בֵּיצִים
  2. ב.שֶׁלִּדְבוֹרִים שׁוֹנוֹת בַּכַּוֶּרֶת יֵשׁ תַּפְקִידִים שׁוֹנִים
  3. ג.שֶׁהַדְּבוֹרִים הַצְּעִירוֹת אוֹסְפוֹת צוּף מִן הַפְּרָחִים
  4. ד.שֶׁהַמַּלְכָּה שׁוֹמֶרֶת עַל הַפֶּתַח שֶׁל הַכַּוֶּרֶת
4. עַל אֵיזוֹ שְׁאֵלָה אֵין תְּשׁוּבָה בַּקֶּטַע?
  1. א.מִמָּה הַדְּבוֹרִים מְכִינוֹת אֶת הַדְּבַשׁ?
  2. ב.כַּמָּה פְּעָמִים יְכוֹלָה דְּבוֹרַת הַדְּבַשׁ לַעֲקוֹץ?
  3. ג.כַּמָּה זְמַן חַיָּה מַלְכַּת הַדְּבוֹרִים?
  4. ד.מָה עוֹשׂוֹת הַדְּבוֹרִים הַצְּעִירוֹת בַּכַּוֶּרֶת?
5. הַשְׁלִימוּ אֶת הַמִּשְׁפָּט לְפִי הַקֶּטַע. כִּתְבוּ מִילָּה אַחַת עַל הַקַּו.
הַדְּבוֹרָה מַרְאָה לַחֲבֵרוֹתֶיהָ הֵיכָן הַפְּרָחִים בְּעֶזְרַת מְיוּחָד.

עִבְרוּ לָעַמּוּד הַבָּא.

קִרְאוּ אֶת הַסִּיפּוּר שֶׁלִּפְנֵיכֶם וַעֲנוּ עַל הַשְּׁאֵלוֹת שֶׁאַחֲרָיו.

יוֹם שִׁישִּׁי בַּשּׁוּק
  1. בְּיוֹם שִׁישִּׁי בַּבּוֹקֶר הָלְכוּ יַעֲלִי, אִימָּא וַעֲמִית הַקָּטָן אֶל הַשּׁוּק.
  2. הַשּׁוּק הָיָה הוֹמֶה: מוֹכְרִים צָעֲקוּ, אֲנָשִׁים נִדְחֲקוּ, וּמוּזִיקָה נִשְׁמְעָה מִכָּל צַד.
  3. "יַעֲלִי, הַחֲזִיקִי לַעֲמִית אֶת הַיָּד," בִּיקְּשָׁה אִימָּא, וְנֶעֶמְדָה בַּתּוֹר לַגְּבִינוֹת.
  4. יַעֲלִי, יַלְדָּה בַּת תֵּשַׁע, אָחֲזָה הֵיטֵב בַּיָּד שֶׁל אָחִיהָ בֶּן הֶחָמֵשׁ.
  5. פִּתְאוֹם רָאֲתָה דּוּכָן שֶׁל צְמִידִים צִבְעוֹנִיִּים, וְהֵרִימָה צָמִיד אֶחָד כְּדֵי לִמְדּוֹד אוֹתוֹ.
  6. רַק לְרֶגַע אֶחָד הִיא שִׁחְרְרָה אֶת הַיָּד שֶׁל עֲמִית.
  7. כְּשֶׁהִסְתּוֹבְבָה, עֲמִית כְּבָר לֹא הָיָה לְיָדָהּ. הַלֵּב שֶׁלָּהּ דָּפַק בְּחוֹזְקָה.
  8. "אִימָּא! עֲמִית נֶעֱלַם!" קָרְאָה יַעֲלִי. אִימָּא הֶחֱוִירָה וְעָזְבָה מִיָּד אֶת הַתּוֹר.
  9. "צָרִיךְ לָרוּץ וּלְחַפֵּשׂ אוֹתוֹ בְּכָל הַשּׁוּק!" אָמְרָה אִימָּא בְּבֶהָלָה.
  10. גַּם יַעֲלִי רָצְתָה לָרוּץ, אֲבָל אָז נִזְכְּרָה בַּכְּלָל שֶׁאִימָּא לִימְּדָה אוֹתָם.
  11. "אִימָּא, חַכִּי," אָמְרָה. "אַתְּ בְּעַצְמֵךְ לִימַּדְתְּ אוֹתָנוּ מָה עוֹשִׂים כְּשֶׁהוֹלְכִים לְאִיבּוּד:
  12. עוֹמְדִים בַּמָּקוֹם, לֹא זָזִים, וּמְבַקְּשִׁים עֶזְרָה מִמּוֹכֵר. עֲמִית בֶּטַח זוֹכֵר!"
  13. אִימָּא נָשְׁמָה נְשִׁימָה עֲמוּקָּה. "אַתְּ צוֹדֶקֶת," אָמְרָה. "נְבַקֵּשׁ עֶזְרָה גַּם אֲנַחְנוּ."
  14. הֵן נִיגְּשׁוּ אֶל מוֹכֵר הַיְּרָקוֹת, וְסִיפְּרוּ לוֹ שֶׁיֶּלֶד קָטָן בְּחוּלְצָה צְהוּבָּה נֶעֱלַם.
  15. "אַל תִּדְאֲגוּ," אָמַר הַמּוֹכֵר. "כָּל הַמּוֹכְרִים כָּאן מַכִּירִים זֶה אֶת זֶה."
  16. הוּא הוֹצִיא טֶלֶפוֹן וְהִתְקַשֵּׁר אֶל חֲבֵרָיו בַּדּוּכָנִים הָאֲחֵרִים.
  17. אַחֲרֵי דַּקָּה אֲרוּכָּה הוּא חִיֵּיךְ: "מָצְאוּ אוֹתוֹ! הוּא עוֹמֵד לְיַד דּוּכַן הַמִּיצִים."
  18. אִימָּא וְיַעֲלִי מִיהֲרוּ לְשָׁם. עֲמִית עָמַד בְּשֶׁקֶט לְיַד מוֹכֶרֶת הַמִּיצִים, וְלֹא זָז.
  19. "רָאִיתִי בָּלוֹן אָדוֹם וְהָלַכְתִּי אַחֲרָיו," סִיפֵּר, "וּפִתְאוֹם לֹא רָאִיתִי אֶתְכֶן.
  20. אָז עָמַדְתִּי בַּמָּקוֹם וְאָמַרְתִּי לַמּוֹכֶרֶת שֶׁהָלַכְתִּי לְאִיבּוּד. אֲפִילּוּ לֹא בָּכִיתִי!"
  21. אִימָּא חִיבְּקָה אוֹתוֹ חָזָק. יַעֲלִי הִשְׁפִּילָה אֶת עֵינֶיהָ: "סְלִיחָה שֶׁעָזַבְתִּי אֶת הַיָּד."
  22. "כּוּלָּנוּ טוֹעִים לִפְעָמִים," אָמְרָה אִימָּא. "אֲבָל הַיּוֹם שְׁנֵיכֶם זְכַרְתֶּם מָה שֶׁחָשׁוּב."
  23. מֵאָז אוֹתוֹ יוֹם, יַעֲלִי מַחֲזִיקָה לַעֲמִית אֶת הַיָּד גַּם כְּשֶׁיֵּשׁ צְמִידִים יָפִים.

עִבְרוּ לָעַמּוּד הַבָּא.

כָּעֵת, עֲנוּ עַל הַשְּׁאֵלוֹת שֶׁלִּפְנֵיכֶם.
הַקִּיפוּ אֶת הַתְּשׁוּבָה הַנְּכוֹנָה בְּכָל שְׁאֵלָה.

1. מָה פֵּירוּשׁ הַמִּילָה "הוֹמֶה" (שׁוּרָה 2)?
  1. א.שָׁקֵט וְרֵיק
  2. ב.קָטָן וְצַר
  3. ג.נָקִי וּמְסוּדָּר
  4. ד.מָלֵא רַעַשׁ וּתְנוּעָה
2. בְּשׁוּרָה 6 כָּתוּב: "רַק לְרֶגַע אֶחָד הִיא שִׁחְרְרָה אֶת הַיָּד שֶׁל עֲמִית".
הַמַּטָּרָה שֶׁל הַמִּשְׁפָּט הַזֶּה הִיא –
  1. א.לְהַסְבִּיר שֶׁיַּעֲלִי אֵינָהּ אוֹהֶבֶת לְהַחֲזִיק יָד
  2. ב.לְהַדְגִּישׁ שֶׁעֲמִית נֶעֱלַם בְּתוֹךְ זְמַן קָצָר מְאוֹד
  3. ג.לְסַפֵּר שֶׁאִימָּא הִרְשְׁתָה לְיַעֲלִי לַעֲזוֹב אֶת הַיָּד
