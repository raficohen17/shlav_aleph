חֵלֶק רִאשׁוֹן: קִטְעֵי קְרִיאָה וּשְׁאֵלוֹת

קִרְאוּ אֶת קֶטַע הַמֵּידָע שֶׁלִּפְנֵיכֶם וַעֲנוּ עַל הַשְּׁאֵלוֹת שֶׁאַחֲרָיו.

הַדּוֹלְפִין
  1. הַדּוֹלְפִין חַי בַּיָּם וְשׂוֹחֶה כְּמוֹ דָּג, אֲבָל הוּא בִּכְלָל אֵינוֹ דָּג.
  2. הַדּוֹלְפִין הוּא יוֹנֵק: הָאִימָּא יוֹלֶדֶת גּוּר וּמֵינִיקָה אוֹתוֹ בְּחָלָב.
  3. דָּגִים נוֹשְׁמִים בְּתוֹךְ הַמַּיִם, אֲבָל הַדּוֹלְפִין נוֹשֵׁם אֲוִויר, כָּמוֹנוּ.
  4. לָכֵן הוּא חַיָּיב לַעֲלוֹת אֶל פְּנֵי הַמַּיִם כָּל כַּמָּה דַּקּוֹת.
  5. בְּרֹאשׁ הַדּוֹלְפִין יֵשׁ נְחִיר, פֶּתַח קָטָן שֶׁדַּרְכּוֹ הוּא שׁוֹאֵף אֲוִויר.
  6. כְּשֶׁהַדּוֹלְפִין צוֹלֵל, הַנְּחִיר נִסְגָּר הֵיטֵב וּמַיִם אֵינָם נִכְנָסִים לְתוֹכוֹ.
  7. אֲבָל אֵיךְ הַדּוֹלְפִין יָשֵׁן? הֲרֵי אִם יֵירָדֵם לְגַמְרֵי, לֹא יַעֲלֶה לִנְשׁוֹם!
  8. חוֹקְרִים גִּילּוּ שֶׁהוּא יָשֵׁן בְּדֶרֶךְ מְיוּחֶדֶת: רַק חֲצִי מֵהַמּוֹחַ שֶׁלּוֹ נִרְדָּם.
  9. הַחֵצִי הַשֵּׁנִי נִשְׁאָר עֵר, וְדוֹאֵג שֶׁהַדּוֹלְפִין יַעֲלֶה לִנְשׁוֹם בַּזְּמַן.
  10. בִּזְמַן הַשֵּׁינָה גַּם עַיִן אַחַת נִשְׁאֶרֶת פְּקוּחָה, וְשׁוֹמֶרֶת מִפְּנֵי כְּרִישִׁים.
  11. אַחֲרֵי כִּשְׁעָתַיִים הַחֲצָאִים מִתְחַלְּפִים: הַחֵצִי הָעֵר נִרְדָּם, וְהַיָּשֵׁן מִתְעוֹרֵר.
  12. דּוֹלְפִינִים חַיִּים בִּקְבוּצוֹת, וְהֵם "מְדַבְּרִים" זֶה עִם זֶה בִּשְׁרִיקוֹת.
  13. לְכָל דּוֹלְפִין יֵשׁ שְׁרִיקָה מְיוּחֶדֶת מִשֶּׁלּוֹ, וְהִיא מְשַׁמֶּשֶׁת לוֹ כְּמוֹ שֵׁם.
  14. כְּשֶׁדּוֹלְפִין מִתְרַחֵק מֵהַקְּבוּצָה, חֲבֵרָיו שׁוֹרְקִים אֶת הַשְּׁרִיקָה שֶׁלּוֹ וְקוֹרְאִים לוֹ.
  15. גּוּר צָעִיר לוֹמֵד לְזַהוֹת אֶת שְׁרִיקַת אִימּוֹ כְּבָר בַּיָּמִים הָרִאשׁוֹנִים לְחַיָּיו.
  16. אוּלַי יוֹם אֶחָד נִלְמַד גַּם אֲנַחְנוּ לְהָבִין אֶת שְׂפַת הַשְּׁרִיקוֹת.

עִבְרוּ לָעַמּוּד הַבָּא.

כָּעֵת, עֲנוּ עַל הַשְּׁאֵלוֹת שֶׁלִּפְנֵיכֶם.
הַקִּיפוּ אֶת הַתְּשׁוּבָה הַנְּכוֹנָה בְּכָל שְׁאֵלָה.

1. מַדּוּעַ הַדּוֹלְפִין חַיָּיב לַעֲלוֹת אֶל פְּנֵי הַמַּיִם כָּל כַּמָּה דַּקּוֹת?
  1. א.כְּדֵי לִשְׁמוֹר עַל עַצְמוֹ מִפְּנֵי כְּרִישִׁים
  2. ב.כִּי הוּא נוֹשֵׁם אֲוִויר
  3. ג.כְּדֵי לִשְׁרוֹק לַחֲבֵרָיו שֶׁבַּקְּבוּצָה
  4. ד.כִּי הוּא רוֹצֶה לִישׁוֹן עַל פְּנֵי הַמַּיִם
2. בְּשׁוּרָה 7 כְּתוּבָה שְׁאֵלָה: "אֲבָל אֵיךְ הַדּוֹלְפִין יָשֵׁן?"
הַמַּטָּרָה שֶׁל הַשְּׁאֵלָה הַזֹּאת הִיא –
  1. א.לְהַרְאוֹת שֶׁדּוֹלְפִינִים אֵינָם יְשֵׁנִים אַף פַּעַם
  2. ב.לְסַפֵּר שֶׁהַחוֹקְרִים עֲדַיִין אֵינָם יוֹדְעִים אֶת הַתְּשׁוּבָה
  3. ג.לְהַסְבִּיר מַדּוּעַ דָּגִים נוֹשְׁמִים בְּתוֹךְ הַמַּיִם
  4. ד.לְהַצִּיג בְּעָיָה, שֶׁהַשּׁוּרוֹת הַבָּאוֹת מַסְבִּירוֹת אֵיךְ הַדּוֹלְפִין פּוֹתֵר אוֹתָהּ
3. בְּשׁוּרָה 13 כָּתוּב: "וְהִיא מְשַׁמֶּשֶׁת לוֹ כְּמוֹ שֵׁם".
לְמִי אוֹ לְמָה מִתְיַיחֶסֶת הַמִּילָה "הִיא"?
  1. א.לַשְּׁרִיקָה הַמְּיוּחֶדֶת
  2. ב.לַקְּבוּצָה שֶׁל הַדּוֹלְפִינִים
  3. ג.לָאִימָּא שֶׁל הַגּוּר
  4. ד.לָעַיִן הַפְּקוּחָה
4. עַל אֵיזוֹ שְׁאֵלָה אֵין תְּשׁוּבָה בַּקֶּטַע?
  1. א.דֶּרֶךְ אֵיזֶה חֵלֶק בַּגּוּף הַדּוֹלְפִין שׁוֹאֵף אֲוִויר?
  2. ב.מָה קוֹרֶה לַמּוֹחַ שֶׁל הַדּוֹלְפִין בִּזְמַן הַשֵּׁינָה?
  3. ג.מָה אוֹכֵל הַדּוֹלְפִין?
  4. ד.אֵיךְ דּוֹלְפִינִים קוֹרְאִים לְחָבֵר שֶׁהִתְרַחֵק מֵהַקְּבוּצָה?
5. אֵיזֶה שֵׁם אַחֵר מַתְאִים בְּיוֹתֵר לַקֶּטַע כּוּלּוֹ?
  1. א.הַדָּג הֶחָכָם בְּיוֹתֵר בַּיָּם
  2. ב.יוֹנֵק מְיוּחָד בַּיָּם
  3. ג.אֵיךְ יְשֵׁנִים הַדּוֹלְפִינִים
  4. ד.כְּרִישִׁים וְדוֹלְפִינִים

עִבְרוּ לָעַמּוּד הַבָּא.

קִרְאוּ אֶת הַסִּיפּוּר שֶׁלִּפְנֵיכֶם וַעֲנוּ עַל הַשְּׁאֵלוֹת שֶׁאַחֲרָיו.

שְׁאֵלָה אַחַת קְטַנָּה
  1. בְּיוֹם שְׁלִישִׁי אַחַר הַצָּהֳרַיִים הָלַךְ עוֹמֶר עִם סַבָּא בֶּנִי לַסִּפְרִיָּיה הָעִירוֹנִית.
  2. עוֹמֶר, יֶלֶד בֶּן שְׁמוֹנֶה, חִיפֵּשׂ סֵפֶר עַל כּוֹכָבִים בִּשְׁבִיל עֲבוֹדָה לַכִּיתָּה.
  3. סַבָּא הִתְיַישֵּׁב עַל כּוּרְסָה לְיַד הַחַלּוֹן, וּפָתַח אֶת הָעִיתּוֹן שֶׁלּוֹ.
  4. עוֹמֶר שׁוֹטֵט בֵּין הַמַּדָּפִים הַגְּבוֹהִים. הָיוּ שָׁם אַלְפֵי סְפָרִים!
  5. הוּא חִיפֵּשׂ בְּמַדָּף אֶחָד, וְאַחַר כָּךְ בְּמַדָּף אַחֵר, אֲבָל לֹא מָצָא דָּבָר.
  6. לְיַד הַדֶּלְפֵּק יָשְׁבָה הַסַּפְרָנִית וְהִקְלִידָה בַּמַּחְשֵׁב. עוֹמֶר הֵצִיץ בָּהּ מֵרָחוֹק.
  7. "וּמָה אִם הִיא עֲסוּקָה? וּמָה אִם הַשְּׁאֵלָה שֶׁלִּי טִיפְּשִׁית?" חָשַׁב, וְלֹא נִיגַּשׁ.
  8. אַחֲרֵי חֲצִי שָׁעָה חָזַר עוֹמֶר אֶל סַבָּא בְּיָדַיִים רֵיקוֹת.
  9. "אֵין כָּאן סֵפֶר עַל כּוֹכָבִים," מִלְמֵל. "בּוֹא נַחֲזוֹר הַבַּיְתָה."
  10. סַבָּא קִיפֵּל אֶת הָעִיתּוֹן. "שָׁאַלְתָּ אֶת הַסַּפְרָנִית?"
  11. עוֹמֶר הִשְׁפִּיל אֶת עֵינָיו. "לֹא נָעִים לִי. אֲנִי אָמוּר לִמְצוֹא לְבַד."
  12. "אֲנִי כְּבָר בֶּן שִׁבְעִים," אָמַר סַבָּא, "וַאֲפִילּוּ אֲנִי שׁוֹאֵל שְׁאֵלוֹת כָּל יוֹם.
  13. רַק הַבּוֹקֶר שָׁאַלְתִּי אֶת הַנֶּהָג אֵיפֹה לָרֶדֶת. מִי שֶׁשּׁוֹאֵל – אֵינוֹ הוֹלֵךְ לְאִיבּוּד."
  14. עוֹמֶר הִיסֵּס, אֲבָל לְבַסּוֹף נִיגַּשׁ אֶל הַדֶּלְפֵּק בִּצְעָדִים קְטַנִּים.
  15. "סְלִיחָה," אָמַר בְּקוֹל חָלוּשׁ, "אֵיפֹה יֵשׁ סְפָרִים עַל כּוֹכָבִים?"
  16. הַסַּפְרָנִית הֵרִימָה אֶת רֹאשָׁהּ וְחִיְּיכָה. "אֵיזוֹ שְׁאֵלָה מְצוּיֶּנֶת! בּוֹא אִיתִּי."
  17. הִיא הוֹבִילָה אוֹתוֹ אֶל מַדָּף בַּפִּינָּה, וְשָׁלְפָה מִמֶּנּוּ שְׁלוֹשָׁה סְפָרִים עָבִים.
  18. "בְּדִיּוּק הַיּוֹם קִיבַּלְנוּ סֵפֶר חָדָשׁ עַל כּוֹכָבִים. אַתָּה הָרִאשׁוֹן שֶׁקּוֹרֵא בּוֹ!"
  19. עוֹמֶר חִיבֵּק אֶת הַסְּפָרִים וְרָץ אֶל סַבָּא. פָּנָיו זָרְחוּ.
  20. בַּדֶּרֶךְ הַחוּצָה נֶעֱצַר עוֹמֶר לְיַד הַדֶּלְפֵּק. "אֶפְשָׁר לִשְׁאוֹל עוֹד שְׁאֵלָה?"
  21. "בְּוַודַּאי," עָנְתָה הַסַּפְרָנִית.
  22. "יֵשׁ לָכֶם גַּם סְפָרִים עַל דִּינוֹזָאוּרִים?" שָׁאַל עוֹמֶר, וְסַבָּא צָחַק.

עִבְרוּ לָעַמּוּד הַבָּא.

כָּעֵת, עֲנוּ עַל הַשְּׁאֵלוֹת שֶׁלִּפְנֵיכֶם.
הַקִּיפוּ אֶת הַתְּשׁוּבָה הַנְּכוֹנָה בְּכָל שְׁאֵלָה.

1. מַדּוּעַ עוֹמֶר לֹא נִיגַּשׁ אֶל הַסַּפְרָנִית כְּשֶׁרָאָה אוֹתָהּ לְיַד הַדֶּלְפֵּק (שׁוּרוֹת 6–7)?
  1. א.כִּי סַבָּא בִּיקֵּשׁ מִמֶּנּוּ לִמְצוֹא אֶת הַסֵּפֶר לְבַד
  2. ב.כִּי כְּבָר מָצָא סֵפֶר עַל כּוֹכָבִים
  3. ג.כִּי הַסַּפְרָנִית אָמְרָה לוֹ שֶׁהִיא עֲסוּקָה
  4. ד.כִּי חָשַׁשׁ שֶׁהִיא עֲסוּקָה וְשֶׁהַשְּׁאֵלָה שֶׁלּוֹ טִיפְּשִׁית
2. מָה פֵּירוּשׁ הַמִּילָה "שׁוֹטֵט" (שׁוּרָה 4)?
  1. א.הִסְתּוֹבֵב מִמָּקוֹם לְמָקוֹם
  2. ב.הִתְחַבֵּא
  3. ג.קָרָא בְּקוֹל רָם
  4. ד.טִיפֵּס לְמַעְלָה
3. מָה קָרָה לִפְנֵי שֶׁעוֹמֶר חָזַר אֶל סַבָּא בְּיָדַיִים רֵיקוֹת?
  1. א.סַבָּא סִיפֵּר שֶׁשָּׁאַל אֶת הַנֶּהָג אֵיפֹה לָרֶדֶת
  2. ב.הַסַּפְרָנִית שָׁלְפָה שְׁלוֹשָׁה סְפָרִים מֵהַמַּדָּף
  3. ג.עוֹמֶר הֵצִיץ בַּסַּפְרָנִית מֵרָחוֹק
  4. ד.עוֹמֶר שָׁאַל אִם יֵשׁ סְפָרִים עַל דִּינוֹזָאוּרִים
4. אֵיזוֹ מִילָה יְכוֹלָה לְהַחֲלִיף אֶת הַמִּילָה "הִיסֵּס" (שׁוּרָה 14)?
  1. א.מִיהֵר
