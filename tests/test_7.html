חֵלֶק רִאשׁוֹן: קִטְעֵי קְרִיאָה וּשְׁאֵלוֹת

קִרְאוּ אֶת הַהוֹרָאוֹת שֶׁלִּפְנֵיכֶם וַעֲנוּ עַל הַשְּׁאֵלוֹת שֶׁאַחֲרֵיהֶן.

אֵיךְ מְכִינִים מַאֲכִיל צִיפּוֹרִים מִבַּקְבּוּק?
  1. בַּחוֹרֶף קָשֶׁה לַצִּיפּוֹרִים לִמְצוֹא אוֹכֶל. רוֹצִים לַעֲזוֹר לָהֶן?
  2. אֶפְשָׁר לְהָכִין לָהֶן מַאֲכִיל פָּשׁוּט, אֲפִילּוּ מִבַּקְבּוּק שֶׁעָמַדְתֶּם לִזְרוֹק.
  3. מָה צָרִיךְ לְהָכִין? בַּקְבּוּק פְּלַסְטִיק רֵיק עִם פְּקָק, כַּף עֵץ אֲרוּכָּה,
  4. חוּט חָזָק, מִסְפָּרַיִים וְשַׂקִּית שֶׁל זְרָעִים. וְעַכְשָׁיו – לַעֲבוֹדָה!
  5. שָׁלָב 1: שׁוֹטְפִים אֶת הַבַּקְבּוּק הֵיטֵב וּמְיַיבְּשִׁים אוֹתוֹ לְגַמְרֵי.
  6. שָׁלָב 2: חוֹתְכִים בַּבַּקְבּוּק שְׁנֵי חוֹרִים זֶה מוּל זֶה – אֶחָד קָטָן וְאֶחָד גָּדוֹל.
  7. זִכְרוּ: חוֹתְכִים רַק יַחַד עִם מְבוּגָּר, כִּי הַמִּסְפָּרַיִים חַדִּים!
  8. שָׁלָב 3: מַשְׁחִילִים כַּף עֵץ דֶּרֶךְ הַחוֹרִים. מֵהַחוֹר הַגָּדוֹל יִפְּלוּ זְרָעִים לַכַּף.
  9. שָׁלָב 4: מְמַלְּאִים אֶת הַבַּקְבּוּק בִּזְרָעִים וְסוֹגְרִים אוֹתוֹ בַּפְּקָק.
  10. שָׁלָב 5: קוֹשְׁרִים חוּט סְבִיב צַוַּואר הַבַּקְבּוּק וְתוֹלִים אוֹתוֹ עַל עָנָף.
  11. כַּמָּה עֵצוֹת חֲשׁוּבוֹת: תְּלוּ אֶת הַמַּאֲכִיל בְּמָקוֹם גָּבוֹהַּ וּמוּצָל.
  12. בְּמָקוֹם גָּבוֹהַּ הַצִּיפּוֹרִים מוּגַנּוֹת מִפְּנֵי חֲתוּלִים,
  13. וּבְמָקוֹם מוּצָל הַזְּרָעִים אֵינָם מִתְקַלְקְלִים מֵהַחוֹם.
  14. בִּדְקוּ אֶת הַמַּאֲכִיל כָּל כַּמָּה יָמִים. אִם הַזְּרָעִים אוֹזְלִים, מַלְּאוּ אוֹתוֹ שׁוּב.
  15. אַל תִּתְאַכְזְבוּ אִם בַּיָּמִים הָרִאשׁוֹנִים אַף צִיפּוֹר לֹא תָּבוֹא.
  16. הַצִּיפּוֹרִים חַשְׁדָנִיּוֹת, וְהֵן צְרִיכוֹת זְמַן כְּדֵי לְהִתְרַגֵּל לְדָבָר חָדָשׁ.
  17. חַכּוּ בְּסַבְלָנוּת, וּבְקָרוֹב תִּרְאוּ אוֹרְחוֹת מְצַיְּיצוֹת לְיַד הַחַלּוֹן.

עִבְרוּ לָעַמּוּד הַבָּא.

כָּעֵת, עֲנוּ עַל הַשְּׁאֵלוֹת שֶׁלִּפְנֵיכֶם.
הַקִּיפוּ אֶת הַתְּשׁוּבָה הַנְּכוֹנָה בְּכָל שְׁאֵלָה.

1. מָה עוֹשִׂים מִיָּד לִפְנֵי שֶׁמְּמַלְּאִים אֶת הַבַּקְבּוּק בִּזְרָעִים?
  1. א.קוֹשְׁרִים חוּט סְבִיב צַוַּואר הַבַּקְבּוּק
  2. ב.שׁוֹטְפִים אֶת הַבַּקְבּוּק וּמְיַיבְּשִׁים אוֹתוֹ
  3. ג.מַשְׁחִילִים אֶת כַּף הָעֵץ דֶּרֶךְ הַחוֹרִים
  4. ד.תּוֹלִים אֶת הַבַּקְבּוּק עַל עָנָף
2. מָה פֵּירוּשׁ הַמִּילָּה "אוֹזְלִים" (שׁוּרָה 14)?
  1. א.נִגְמָרִים
  2. ב.נִרְטָבִים
  3. ג.צוֹמְחִים
  4. ד.מַבְרִיקִים
3. יַלְדָּה הֵכִינָה מַאֲכִיל צִיפּוֹרִים לְפִי הַהוֹרָאוֹת שֶׁבַּקֶּטַע.
אֵיפֹה כְּדַאי לָהּ לִתְלוֹת אוֹתוֹ?
  1. א.עַל גָּדֵר נְמוּכָה, בְּמָקוֹם מוּצָל
  2. ב.עַל עָנָף גָּבוֹהַּ, בְּצֵל הָעֵץ
  3. ג.עַל עַמּוּד גָּבוֹהַּ, בְּמָקוֹם שֶׁהַשֶּׁמֶשׁ זוֹרַחַת עָלָיו כָּל הַיּוֹם
  4. ד.עַל רִצְפַּת הַמִּרְפֶּסֶת, בְּמָקוֹם שֶׁהַשֶּׁמֶשׁ זוֹרַחַת עָלָיו
4. בְּשׁוּרָה 7 כָּתוּב: "כִּי הַמִּסְפָּרַיִים חַדִּים!"
הַמַּטָּרָה שֶׁל הַמִּילִּים הָאֵלֶּה הִיא –
  1. א.לְסַפֵּר אֵילוּ מִסְפָּרַיִים כְּדַאי לִקְנוֹת
  2. ב.לְשַׁכְנֵעַ אוֹתָנוּ לֹא לְהָכִין מַאֲכִיל צִיפּוֹרִים
  3. ג.לְהַדְגִּישׁ שֶׁהַבַּקְבּוּק עָשׂוּי מִפְּלַסְטִיק קָשֶׁה
  4. ד.לְהַסְבִּיר מַדּוּעַ חוֹתְכִים רַק יַחַד עִם מְבוּגָּר
5. בְּשׁוּרָה 2 כָּתוּב: "אֲפִילּוּ מִבַּקְבּוּק שֶׁעָמַדְתֶּם לִזְרוֹק".
מַדּוּעַ כָּתוּב "אֲפִילּוּ"?
  1. א.כְּדֵי לְהַדְגִּישׁ שֶׁלֹּא צָרִיךְ חוֹמָרִים מְיוּחָדִים – גַּם בַּקְבּוּק יָשָׁן מַתְאִים
  2. ב.כִּי צִיפּוֹרִים אוֹהֲבוֹת בַּקְבּוּקִים יוֹתֵר מִכָּל דָּבָר אַחֵר
  3. ג.כִּי אָסוּר לִזְרוֹק בַּקְבּוּקִים לַפַּח
  4. ד.כִּי מַאֲכִיל מִבַּקְבּוּק טוֹב יוֹתֵר מִמַּאֲכִיל מֵעֵץ

עִבְרוּ לָעַמּוּד הַבָּא.

קִרְאוּ אֶת הַסִּיפּוּר שֶׁלִּפְנֵיכֶם וַעֲנוּ עַל הַשְּׁאֵלוֹת שֶׁאַחֲרָיו.

הַשְּׁתִיקָה שֶׁל רוֹעִי
  1. בַּבּוֹקֶר רָאָה אָדָם אֶת רוֹעִי, חֲבֵרוֹ הַטּוֹב, בִּשְׁעַר בֵּית הַסֵּפֶר.
  2. רוֹעִי עָטַף אֶת צַוָּוארוֹ בְּצָעִיף עָבֶה, אַף עַל פִּי שֶׁהַיּוֹם הָיָה חָמִים.
  3. "בּוֹקֶר טוֹב, רוֹעִי! מָה שְׁלוֹמְךָ?" קָרָא אָדָם.
  4. רוֹעִי לֹא עָנָה. הוּא הִמְשִׁיךְ לָלֶכֶת בְּלִי לְהִסְתּוֹבֵב.
  5. "מוּזָר," חָשַׁב אָדָם. "אוּלַי הוּא לֹא שָׁמַע אוֹתִי."
  6. בַּהַפְסָקָה שָׁאַל אָדָם: "בָּא לְךָ לְשַׂחֵק כַּדּוּרֶגֶל?"
  7. רוֹעִי הִצְבִּיעַ עַל גְּרוֹנוֹ, מָשַׁךְ בִּכְתֵפָיו וְנִשְׁאַר לָשֶׁבֶת עַל הַסַּפְסָל.
  8. אָדָם לֹא הֵבִין אֶת הַתְּנוּעָה. לִיבּוֹ הִתְכַּוֵּוץ.
  9. "הוּא בֶּטַח כּוֹעֵס עָלַיי," חָשַׁב. "אֶתְמוֹל לֹא מָסַרְתִּי לוֹ אֶת הַכַּדּוּר,
  10. וְהַקְּבוּצָה שֶׁלָּנוּ הִפְסִידָה. עַכְשָׁיו הוּא לֹא רוֹצֶה לְדַבֵּר אִיתִּי."
  11. גַּם בַּאֲרוּחַת הָעֶשֶׂר יָשַׁב רוֹעִי בְּשֶׁקֶט וְלֹא אָמַר אַף מִילָּה.
  12. כָּל הַיּוֹם הִסְתּוֹבֵב אָדָם עָצוּב. הוּא אֲפִילּוּ לֹא נָגַע בַּכָּרִיךְ שֶׁלּוֹ.
  13. בְּסוֹף הַיּוֹם אָזַר אָדָם אוֹמֶץ וְנִיגַּשׁ אֶל רוֹעִי.
  14. "סְלִיחָה שֶׁלֹּא מָסַרְתִּי לְךָ אֶת הַכַּדּוּר. אַל תִּכְעַס עָלַיי," אָמַר בְּקוֹל רוֹעֵד.
  15. רוֹעִי פָּקַח עֵינַיִים גְּדוֹלוֹת, כָּתַב מַשֶּׁהוּ בְּמַחְבֶּרֶת וְהוֹשִׁיט אוֹתָהּ לְאָדָם.
  16. אָדָם קָרָא: "אֲנִי לֹא כּוֹעֵס בִּכְלָל! יֵשׁ לִי דַּלֶּקֶת בַּגָּרוֹן,
  17. וְהַקּוֹל שֶׁלִּי נֶעֱלַם. הָרוֹפֵא אָמַר לִי לֹא לְדַבֵּר שְׁלוֹשָׁה יָמִים."
  18. אָדָם הִבִּיט בַּצָּעִיף, נִזְכַּר בַּיָּד שֶׁהִצְבִּיעָה עַל הַגָּרוֹן, וּפָרַץ בִּצְחוֹק.
  19. "וַאֲנִי דָּאַגְתִּי כָּל הַיּוֹם לְחִינָּם!" אָמַר. "לָמָּה לֹא כָּתַבְתָּ לִי בַּבּוֹקֶר?"
  20. רוֹעִי חִיֵּיךְ וְכָתַב: "לֹא חָשַׁבְתִּי עַל זֶה. וְלָמָּה אַתָּה לֹא שָׁאַלְתָּ?"
  21. "צוֹדֵק," אָמַר אָדָם. "בַּפַּעַם הַבָּאָה אֶשְׁאַל, וְלֹא אֲנַחֵשׁ לְבַד."
  22. מֵאוֹתוֹ יוֹם, כְּשֶׁמַּשֶּׁהוּ מַטְרִיד אֶת אָדָם, הוּא קוֹדֶם כּוֹל שׁוֹאֵל.

עִבְרוּ לָעַמּוּד הַבָּא.

כָּעֵת, עֲנוּ עַל הַשְּׁאֵלוֹת שֶׁלִּפְנֵיכֶם.
הַקִּיפוּ אֶת הַתְּשׁוּבָה הַנְּכוֹנָה בְּכָל שְׁאֵלָה.

1. מַדּוּעַ חָשַׁב אָדָם שֶׁרוֹעִי כּוֹעֵס עָלָיו?
  1. א.כִּי רוֹעִי אָמַר לוֹ בַּבּוֹקֶר שֶׁהוּא כּוֹעֵס
  2. ב.כִּי יוֹם קוֹדֶם לֹא מָסַר לְרוֹעִי אֶת הַכַּדּוּר, וְהַקְּבוּצָה הִפְסִידָה
  3. ג.כִּי שָׁכַח לְהָבִיא לְרוֹעִי כָּרִיךְ לַאֲרוּחַת הָעֶשֶׂר
  4. ד.כִּי לֹא הִזְמִין אֶת רוֹעִי לְשַׂחֵק כַּדּוּרֶגֶל בַּהַפְסָקָה
2. בְּשׁוּרָה 15 כָּתוּב: "וְהוֹשִׁיט אוֹתָהּ לְאָדָם".
אֵיזוֹ מִילָּה יְכוֹלָה לְהַחְלִיף אֶת הַמִּילָּה "הוֹשִׁיט"?
  1. א.הֶחְבִּיא
  2. ב.קָרַע
  3. ג.סָגַר
  4. ד.הִגִּישׁ
3. מָה פֵּירוּשׁ הַמִּילָּה "לְחִינָּם" (שׁוּרָה 19)?
  1. א.בְּלִי סִיבָּה אֲמִיתִּית
  2. ב.בְּלִי לְשַׁלֵּם כֶּסֶף
  3. ג.בְּקוֹל רָם
  4. ד.לִזְמַן קָצָר
4. מַדּוּעַ בֶּאֱמֶת לֹא עָנָה רוֹעִי לְאָדָם כָּל הַיּוֹם?
  1. א.כִּי לֹא שָׁמַע אֶת מָה שֶׁאָדָם אָמַר לוֹ
  2. ב.כִּי כָּעַס עַל אָדָם שֶׁלֹּא מָסַר לוֹ אֶת הַכַּדּוּר

קטע 1 – איך מכינים מאכיל ציפורים מבקבוק? (טקסט הוראות)
1גמילוי בזרעים = שלב 4 (ש' 9); מיד לפניו שלב 3 – משחילים את הכף (ש' 8). ב' = שלב 1 (הרבה לפני), א' ו-ד' = שלב 5 (אחרי). טריק: סדר שלבים – למצוא את השלב ולקרוא את המספר שלפניו; לשים לב ל"לפני"/"אחרי".
2א"אם הזרעים אוזלים, מלאו אותו שוב" – ממלאים שוב כשנגמר. טריק: לקרוא את כל המשפט שסביב המילה.
3בש' 11: "במקום גבוה ומוצל" – צריך את שני התנאים. א' = מוצל אבל נמוך (חתולים), ג' = גבוה אבל בשמש (הזרעים מתקלקלים), ד' = לא זה ולא זה. טריק: שאלת יישום – לחפש את ההמלצה ולבדוק כל תנאי בנפרד.

5א"אפילו" מדגיש דבר לא צפוי: גם מבקבוק שנראה כמו אשפה אפשר להכין מאכיל. טריק: ב"אפילו" שואלים "מה מפתיע כאן?".
