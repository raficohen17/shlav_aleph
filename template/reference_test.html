חֵלֶק רִאשׁוֹן: קִטְעֵי קְרִיאָה וּשְׁאֵלוֹת

קִרְאוּ אֶת קֶטַע הַמֵּידָע שֶׁלִּפְנֵיכֶם וַעֲנוּ עַל הַשְּׁאֵלוֹת שֶׁאַחֲרָיו.

הַתְּמָנוּן
  1. בְּעוֹמֶק הַיָּם חַי יְצוּר מוּזָר וּמְיוּחָד מְאוֹד – הַתְּמָנוּן. לַתְּמָנוּן יֵשׁ שְׁמוֹנֶה זְרוֹעוֹת
  2. אֲרוּכּוֹת וּגְמִישׁוֹת, וְעַל כָּל זְרוֹעַ יֵשׁ שׁוּרוֹת שֶׁל כַּפְתּוֹרֵי הַצְמָדָה קְטַנִּים.
  3. בְּעֶזְרַת הַכַּפְתּוֹרִים הָאֵלֶּה הַתְּמָנוּן נִצְמָד לַסְּלָעִים וְאוֹחֵז בְּחוֹזְקָה בַּמָּזוֹן שֶׁלּוֹ.
  4. רוֹב בַּעֲלֵי הַחַיִּים שֶׁאֲנַחְנוּ מַכִּירִים חַיִּים עִם לֵב אֶחָד בִּלְבַד. לַתְּמָנוּן יֵשׁ
  5. שְׁלוֹשָׁה לְבָבוֹת! גַּם הַדָּם שֶׁלּוֹ שׁוֹנֶה מִשֶּׁלָּנוּ: הוּא אֵינוֹ אָדוֹם אֶלָּא כָּחוֹל.
  6. לַתְּמָנוּן אֵין עֲצָמוֹת בַּגּוּף כְּלָל. לָכֵן הוּא יָכוֹל לְהִידָּחֵק וְלַעֲבוֹר דֶּרֶךְ סְדָקִים
  7. צָרִים מְאוֹד בֵּין הַסְּלָעִים. תְּמָנוּן גָּדוֹל יָכוֹל לְהִיכָּנֵס אֲפִילּוּ לְתוֹךְ בַּקְבּוּק קָטָן!
  8. בַּיָּם יֵשׁ דָּגִים גְּדוֹלִים שֶׁרוֹצִים לִטְרוֹף אֶת הַתְּמָנוּן. אֵיךְ הוּא מִתְגּוֹנֵן מִפְּנֵיהֶם?
  9. קוֹדֶם כּוֹל, הַתְּמָנוּן יוֹדֵעַ לְהַסְווֹת אֶת עַצְמוֹ. בְּתוֹךְ רֶגַע הוּא מְשַׁנֶּה אֶת צֶבַע
  10. הָעוֹר שֶׁלּוֹ, עַד שֶׁהוּא נִרְאֶה בְּדִיּוּק כְּמוֹ הַסֶּלַע אוֹ הַחוֹל שֶׁלְּיָדוֹ.
  11. אִם דָּג טוֹרֵף בְּכָל זֹאת מְגַלֶּה אוֹתוֹ, הַתְּמָנוּן מַתִּיז לְעֶבְרוֹ עָנָן שֶׁל דְּיוֹ כֵּהֶה.
  12. הַדָּג אֵינוֹ רוֹאֶה דָּבָר בְּתוֹךְ הֶעָנָן, וּבֵינְתַיִים הַתְּמָנוּן שׂוֹחֶה מִשָּׁם בִּמְהִירוּת.
  13. חוֹקְרִים גִּילּוּ שֶׁהַתְּמָנוּן הוּא גַּם בַּעַל חַיִּים נָבוֹן מְאוֹד. בְּמַעְבָּדָה אַחַת
  14. לָמַד תְּמָנוּן לִפְתּוֹחַ מִכְסֶה שֶׁל צִנְצֶנֶת כְּדֵי לְהַגִּיעַ אֶל הַסַּרְטָן שֶׁבְּתוֹכָהּ.
  15. לְמַעֲשֶׂה, הַתְּמָנוּן פּוֹתֵר בְּעָיוֹת שֶׁגַּם חַיּוֹת גְּדוֹלוֹת מִמֶּנּוּ אֵינָן מַצְלִיחוֹת לִפְתּוֹר.
  16. אָז בַּפַּעַם הַבָּאָה שֶׁתִּרְאוּ תְּמָנוּן, זִכְרוּ: לִפְנֵיכֶם אֶחָד הַיְּצוּרִים הַחֲכָמִים בַּיָּם.

עִבְרוּ לָעַמּוּד הַבָּא.

כָּעֵת, עֲנוּ עַל הַשְּׁאֵלוֹת שֶׁלִּפְנֵיכֶם.
הַקִּיפוּ אֶת הַתְּשׁוּבָה הַנְּכוֹנָה בְּכָל שְׁאֵלָה.

1. מַדּוּעַ הַתְּמָנוּן יָכוֹל לַעֲבוֹר דֶּרֶךְ סְדָקִים צָרִים?
  1. א.כִּי יֵשׁ לוֹ שְׁמוֹנֶה זְרוֹעוֹת אֲרוּכּוֹת
  2. ב.כִּי אֵין לוֹ עֲצָמוֹת בַּגּוּף
  3. ג.כִּי הוּא מְשַׁנֶּה אֶת צֶבַע הָעוֹר שֶׁלּוֹ
  4. ד.כִּי יֵשׁ לוֹ כַּפְתּוֹרֵי הַצְמָדָה קְטַנִּים
2. מָה פֵּירוּשׁ הַמִּילִים "לְהַסְווֹת אֶת עַצְמוֹ" (שׁוּרָה 9)?
  1. א.לְנַקּוֹת אֶת עַצְמוֹ
  2. ב.לְהַאֲכִיל אֶת עַצְמוֹ
  3. ג.לְהַרְדִּים אֶת עַצְמוֹ
  4. ד.לְהַסְתִּיר אֶת עַצְמוֹ
3. בְּשׁוּרָה 7 כָּתוּב שֶׁתְּמָנוּן גָּדוֹל יָכוֹל לְהִיכָּנֵס אֲפִילּוּ לְתוֹךְ בַּקְבּוּק קָטָן.
הַמַּטָּרָה שֶׁל הַמִּשְׁפָּט הַזֶּה הִיא –
  1. א.לְהַדְגִּישׁ עַד כַּמָּה הַגּוּף שֶׁל הַתְּמָנוּן גָּמִישׁ
  2. ב.לְסַפֵּר שֶׁתְּמָנוּנִים אוֹהֲבִים לָגוּר בְּבַקְבּוּקִים
  3. ג.לְהַסְבִּיר אֵיךְ הַתְּמָנוּן בּוֹרֵחַ מִדָּגִים טוֹרְפִים
  4. ד.לְשַׁכְנֵעַ אוֹתָנוּ לֹא לִזְרוֹק בַּקְבּוּקִים לַיָּם
4. בְּשׁוּרָה 13 כָּתוּב שֶׁהַתְּמָנוּן הוּא גַּם בַּעַל חַיִּים נָבוֹן מְאוֹד.
מַדּוּעַ כָּתוּב "גַּם"?
  1. א.כִּי גַּם הַחוֹקְרִים שֶׁבַּמַּעְבָּדָה נְבוֹנִים מְאוֹד
  2. ב.כִּי גַּם הַסַּרְטָן שֶׁבַּצִּנְצֶנֶת הוּא בַּעַל חַיִּים נָבוֹן
  3. ג.כִּי לִפְנֵי כֵן סוּפַּר עַל תְּכוּנוֹת מְיוּחָדוֹת אֲחֵרוֹת שֶׁל הַתְּמָנוּן
  4. ד.כִּי גַּם הַדָּגִים הַטּוֹרְפִים הֵם בַּעֲלֵי חַיִּים נְבוֹנִים
5. עַל אֵיזוֹ שְׁאֵלָה אֵין תְּשׁוּבָה בַּקֶּטַע?
  1. א.כַּמָּה לְבָבוֹת יֵשׁ לַתְּמָנוּן?
  2. ב.כַּמָּה שָׁנִים חַי הַתְּמָנוּן?
  3. ג.מָה צֶבַע הַדָּם שֶׁל הַתְּמָנוּן?
  4. ד.אֵיךְ הַתְּמָנוּן מִתְגּוֹנֵן מִפְּנֵי דָּגִים טוֹרְפִים?

עִבְרוּ לָעַמּוּד הַבָּא.

קִרְאוּ אֶת הַסִּיפּוּר שֶׁלִּפְנֵיכֶם וַעֲנוּ עַל הַשְּׁאֵלוֹת שֶׁאַחֲרָיו.

הָעֲפִיפוֹן שֶׁל אוּרִי
  1. בְּחוּפְשַׁת הַקַּיִץ בָּא אוּרִי, יֶלֶד בֶּן שְׁמוֹנֶה, לְבַקֵּר אֶת סַבָּא יוֹסֵף בַּכְּפָר.
  2. בַּבּוֹקֶר הָרִאשׁוֹן מָצָא אוּרִי בַּמַּחְסָן עֲפִיפוֹן יָשָׁן וּמְאוּבָּק, תָּלוּי עַל הַקִּיר.
  3. "סַבָּא, אֶפְשָׁר לְהָעִיף אוֹתוֹ?" שָׁאַל אוּרִי בְּהִתְלַהֲבוּת.
  4. "הָעֲפִיפוֹן הַזֶּה כְּבָר קָרוּעַ," עָנָה סַבָּא, "אֲבָל אֶפְשָׁר לִבְנוֹת עֲפִיפוֹן חָדָשׁ."
  5. "לִבְנוֹת? זֶה יִימָּשֵׁךְ הֲמוֹן זְמַן," רָטַן אוּרִי. "עָדִיף לִקְנוֹת אֶחָד בַּחֲנוּת."
  6. סַבָּא רַק חִיֵּיךְ, וְהִנִּיחַ עַל הַשּׁוּלְחָן מַקְלוֹת דַּקִּים, נְיָיר צִבְעוֹנִי, דֶּבֶק וְחוּט.
  7. כָּל הַבּוֹקֶר הֵם מָדְדוּ, גָּזְרוּ וְהִדְבִּיקוּ. אוּרִי צִיֵּיר עַל הַנְּיָיר נֶשֶׁר גָּדוֹל.
  8. אַחַר הַצָּהֳרַיִים עָלוּ הַשְּׁנַיִים אֶל הַגִּבְעָה שֶׁמֵּאֲחוֹרֵי הַבַּיִת. הָרוּחַ נָשְׁבָה בְּחוֹזְקָה.
  9. אוּרִי רָץ מַהֵר וְשִׁחְרֵר אֶת הַחוּט. הָעֲפִיפוֹן הִתְרוֹמֵם מְעַט, הִסְתַּחְרֵר וְצָנַח אַרְצָה.
  10. הוּא נִיסָּה שׁוּב וָשׁוּב, אֲבָל בְּכָל פַּעַם הָעֲפִיפוֹן נָפַל אֶל הָאֲדָמָה.
  11. "אָמַרְתִּי לְךָ שֶׁעָדִיף לִקְנוֹת!" קָרָא אוּרִי, וְהִשְׁלִיךְ אֶת הַחוּט בְּכַעַס.
  12. סַבָּא הֵרִים אֶת הָעֲפִיפוֹן וְהִתְבּוֹנֵן בּוֹ. "הָעֲפִיפוֹן הָרִאשׁוֹן שֶׁבָּנִיתִי
  13. נָפַל עֶשְׂרִים פְּעָמִים," אָמַר בְּשֶׁקֶט. "וְאָז הֵבַנְתִּי שֶׁחָסֵר לוֹ זָנָב."
  14. "זָנָב?" הִתְפַּלֵּא אוּרִי. "בִּשְׁבִיל מָה עֲפִיפוֹן צָרִיךְ זָנָב?"
  15. "הַזָּנָב מְיַיצֵּב אֶת הָעֲפִיפוֹן, כְּדֵי שֶׁלֹּא יִסְתַּחְרֵר בָּרוּחַ," הִסְבִּיר סַבָּא.
  16. אוּרִי שָׁתַק רֶגַע. אַחַר כָּךְ רָץ הַבַּיְתָה וְחָזַר עִם סֶרֶט בַּד אָרוֹךְ.
  17. הֵם קָשְׁרוּ אֶת הַסֶּרֶט לִקְצֵה הָעֲפִיפוֹן, וְאוּרִי רָץ שׁוּב בְּמוֹרַד הַגִּבְעָה.
  18. הַפַּעַם הָעֲפִיפוֹן עָלָה גָּבוֹהַּ, גָּבוֹהַּ, עַד שֶׁהַנֶּשֶׁר נִרְאָה כְּמוֹ נְקוּדָּה זְעִירָה בַּשָּׁמַיִים.
  19. אוּרִי אָחַז בַּחוּט בִּשְׁתֵּי יָדָיו, וְעֵינָיו נָצְצוּ.
  20. בָּעֶרֶב שָׁאֲלָה אִימָּא בַּטֶּלֶפוֹן: "מָה נִשְׁמָע? לִקְנוֹת לְךָ עֲפִיפוֹן לְיוֹם הַהוּלֶּדֶת?"
  21. "לֹא צָרִיךְ," עָנָה אוּרִי. "עֲפִיפוֹן מֵהַחֲנוּת כָּל אֶחָד יָכוֹל לְהָעִיף.
  22. אֶת הָעֲפִיפוֹן שֶׁלִּי בָּנִיתִי בְּעַצְמִי – וּמָחָר אֲנִי בּוֹנֶה עוֹד אֶחָד, לְסַבָּא."

עִבְרוּ לָעַמּוּד הַבָּא.

כָּעֵת, עֲנוּ עַל הַשְּׁאֵלוֹת שֶׁלִּפְנֵיכֶם.
הַקִּיפוּ אֶת הַתְּשׁוּבָה הַנְּכוֹנָה בְּכָל שְׁאֵלָה.

1. מַדּוּעַ אוּרִי לֹא רָצָה בַּהַתְחָלָה לִבְנוֹת עֲפִיפוֹן?
  1. א.כִּי הָעֲפִיפוֹן הַיָּשָׁן הָיָה קָרוּעַ
  2. ב.כִּי לְסַבָּא לֹא הָיוּ מַקְלוֹת וּנְיָיר
  3. ג.כִּי חָשַׁב שֶׁהַבְּנִיָּיה תִּימָּשֵׁךְ זְמַן רַב
  4. ד.כִּי פָּחַד שֶׁהָעֲפִיפוֹן יִיפּוֹל
2. מָה פֵּירוּשׁ הַמִּילָה "צָנַח" (שׁוּרָה 9)?
  1. א.נָפַל
  2. ב.נִקְרַע
